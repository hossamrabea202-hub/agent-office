// shared helpers (UE Custom nodes need helper functions inside a struct)
struct JellyFn
{
    float2 h2(float2 p)
    {
        float3 q = frac(float3(p.xyx) * float3(0.1031, 0.1030, 0.0973));
        q += dot(q, q.yzx + 33.33);
        return frac((q.xx + q.yz) * q.zy);
    }
    float3 h3(float3 p)
    {
        p = frac(p * float3(0.1031, 0.1030, 0.0973));
        p += dot(p, p.yxz + 33.33);
        return frac((p.xxy + p.yxx) * p.zyx);
    }
    // 3D cells: x = F1, y = F2 - F1 (small on membrane walls). Feature points drift with tt.
    float2 vor3(float3 x, float tt)
    {
        float3 i = floor(x); float3 f = frac(x);
        float d1 = 8.0; float d2 = 8.0;
        [unroll] for (int z = -1; z <= 1; z++)
        [unroll] for (int y = -1; y <= 1; y++)
        [unroll] for (int k = -1; k <= 1; k++)
        {
            float3 g = float3(k, y, z);
            float3 o = 0.5 + 0.38 * sin(tt + 6.2831853 * h3(i + g));
            float3 r = g + o - f;
            float d = dot(r, r);
            if (d < d1) { d2 = d1; d1 = d; } else if (d < d2) { d2 = d; }
        }
        d1 = sqrt(d1); return float2(d1, sqrt(d2) - d1);
    }
    // 2D cells, periodic in x with period P (wraps around the tentacle girth without a seam)
    float2 vor2(float2 x, float P, float tt)
    {
        float2 i = floor(x); float2 f = frac(x);
        float d1 = 8.0; float d2 = 8.0;
        [unroll] for (int y = -1; y <= 1; y++)
        [unroll] for (int k = -1; k <= 1; k++)
        {
            float2 g = float2(k, y);
            float2 id = i + g; id.x -= P * floor(id.x / P);
            float2 o = 0.5 + 0.38 * sin(tt + 6.2831853 * h2(id));
            float2 r = g + o - f;
            float d = dot(r, r);
            if (d < d1) { d2 = d1; d1 = d; } else if (d < d2) { d2 = d; }
        }
        d1 = sqrt(d1); return float2(d1, sqrt(d2) - d1);
    }
    // value noise, periodic in x with period P
    float vn(float2 p, float P)
    {
        float2 i = floor(p); float2 f = frac(p);
        float2 u = f * f * (3.0 - 2.0 * f);
        float2 i1 = i + 1.0;
        float x0 = i.x - P * floor(i.x / P); float x1 = i1.x - P * floor(i1.x / P);
        float a = h2(float2(x0, i.y)).x; float b = h2(float2(x1, i.y)).x;
        float c = h2(float2(x0, i1.y)).x; float d = h2(float2(x1, i1.y)).x;
        return lerp(lerp(a, b, u.x), lerp(c, d, u.x), u.y);
    }
};

        float br = 0.5 - 0.5 * cos(T * (0.55 + 0.33 * lh) + lh * 6.2831853);   // 7-11 s, eased, no thump
        br = br * br * (3.0 - 2.0 * br);
        return lump * lerp(0.3, 1.0, br) * smoothstep(250.0, 900.0, s) * lerp(1.0, 0.4, smoothstep(8000.0, 11800.0, s));
    }
};
SacFn Sac;
float H = 130.0 / max(Scl, 0.01);                       // ~2.6 m at full swell, fixed (the skin radius is ridged)
float b0 = Sac.B(s, a01L, T);
float bulge = H * b0 * kf;

// gel normal that rides over the sacs: tilt the skinned vertex normal by the bulge slope along the spine and around it,
// so fresnel and wet glints slide across each sac as it breathes (the mesh normal alone made them look painted on)
float dS = 60.0; float dA = 0.004;
float slopeS = H * kf * (Sac.B(s + dS, a01L, T) - Sac.B(s - dS, a01L, T)) / (2.0 * dS);
float slopeA = H * kf * (Sac.B(s, a01L + dA, T) - Sac.B(s, a01L - dA, T)) / (2.0 * dA * 6.2831853 * max(rl, 20.0));
float3 Aw = cross(Tg, Rd);                              // around-the-girth tangent (local)
float3 tiltW = TransformLocalVectorToWorld(Parameters, -(Tg * slopeS + Aw * slopeA));
float3 NwS = normalize((float3)Parameters.TangentToWorld[2]);
GN = normalize(lerp(NwS, RdW, max(cw, 0.85 * smoothstep(1300.0, 2000.0, s) + 0.35)) + tiltW / max(Scl, 0.01) * 1.4);   // sleeve normal: radial blend, no spike facets

// the offset is built in the REF pose but applied after skinning: rotate it by the vertex's skinning rotation (estimated
// as the shortest arc from the pre-skinned to the skinned normal, both in local space), or bent sections push the gel
// sideways into flat sheets.
float3 off = Rd * (pull + g / max(Scl, 0.01) + bulge);
float3 ex = TransformLocalVectorToWorld(Parameters, float3(1, 0, 0));
float3 ey = TransformLocalVectorToWorld(Parameters, float3(0, 1, 0));
float3 ez = TransformLocalVectorToWorld(Parameters, float3(0, 0, 1));
float3 n1 = normalize(float3(dot(NwS, ex), dot(NwS, ey), dot(NwS, ez)));
float3 n0 = normalize(Parameters.PreSkinnedNormal);
float cr = dot(n0, n1);
float3 ax = cross(n0, n1);
float3 offR = off * cr + cross(ax, off) + ax * dot(ax, off) / (1.0 + cr);
return (cr > -0.95) ? offR : off;
