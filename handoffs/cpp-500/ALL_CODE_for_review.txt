// #500: C++ parent for BP_Zombie. Holds the per-Think hot loops natively; BP_Zombie keeps its look, attacks, montages and states.
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Character.h"
#include "ZombieBase.generated.h"

UCLASS(Abstract, Blueprintable)
class CASTLEDEFENDER_API AZombieBase : public ACharacter
{
	GENERATED_BODY()

public:
	AZombieBase();

	// BP_Zombie Think's target pick (nodes CX_501..zfTargetFarmer17), in one native call. Same rules:
	// start from the player; take a soldier that is not dying (LifeSpan < 0.01), within ZombieSoldierHeightRange height and ZombieSoldierRange, if nearer than the current pick;
	// then a farmer that is not Dead, within ZombieFarmerRange, if nearer. Sets the result as Owner (BP_Zombie uses Owner as its target) and returns it.
	UFUNCTION(BlueprintCallable, Category = "Zombie|Brain")
	AActor* PickAndSetTarget(const TArray<AActor*>& Soldiers, const TArray<AActor*>& Farmers);

	// The same, with soldiers and farmers from the crowd registry (no caches on the wave manager needed).
	UFUNCTION(BlueprintCallable, Category = "Zombie|Brain")
	AActor* PickAndSetTargetFromRegistry();

	// Think's "heavy part every Nth tick" divisor: 1 within ZombieNearPlayer, 2 within ZombieMidPlayer, else 4 (K2Node_IfThenElse_11).
	UFUNCTION(BlueprintPure, Category = "Zombie|Brain")
	int32 GetThinkDivisor() const;

	// Near the player (ZombieNearPlayer) or rendered in the last 0.2 s (K2Node_CallFunction_42 / _43).
	UFUNCTION(BlueprintPure, Category = "Zombie|Brain")
	bool IsFullDetail() const;

	// Reads a Blueprint bool variable (for example "Dead") from any actor; false when missing.
	static bool ReadBlueprintBool(const UObject* Object, FName Name);

	// Pure target rule, no world needed (unit-tested in Tests/ZombieBrainTest.cpp).
	struct FCandidate
	{
		FVector Location;
		bool bValid = true;
		bool bDyingOrDead = false;
	};
	// Returns the index of the chosen candidate in Soldiers (0..) or Farmers (Soldiers.Num()..), or INDEX_NONE for "keep the player".
	static int32 ChooseTarget(const FVector& Self, const FVector* Player, const TArray<FCandidate>& Soldiers, const TArray<FCandidate>& Farmers,
		float SoldierRange, float SoldierHeightRange, float FarmerRange);

protected:
	virtual void BeginPlay() override;
	virtual void EndPlay(const EEndPlayReason::Type EndPlayReason) override;

private:
	float DistanceToPlayer() const;
};

bool AZombieBase::ReadBlueprintBool(const UObject* Object, FName Name)
{
	if (!IsValid(Object))
	{
		return false;
	}
	const FBoolProperty* Prop = FindFProperty<FBoolProperty>(Object->GetClass(), Name);
	return Prop && Prop->GetPropertyValue_InContainer(Object);
}

float AZombieBase::DistanceToPlayer() const
{
	const AActor* Player = UGameplayStatics::GetPlayerCharacter(this, 0);
	// GetDistanceTo(None) is 0 in Blueprint: keep that, so a missing player counts as near.
	return Player ? GetDistanceTo(Player) : 0.f;
}

int32 AZombieBase::ChooseTarget(const FVector& Self, const FVector* Player, const TArray<FCandidate>& Soldiers, const TArray<FCandidate>& Farmers,
	float SoldierRange, float SoldierHeightRange, float FarmerRange)
{
	int32 Best = INDEX_NONE;
	float BestDist = Player ? FVector::Dist(Self, *Player) : 0.f;
	for (int32 i = 0; i < Soldiers.Num(); ++i)
	{
		const FCandidate& S = Soldiers[i];
		if (!S.bValid || S.bDyingOrDead || FMath::Abs(S.Location.Z - Self.Z) >= SoldierHeightRange)
		{
			continue;
		}
		const float Dist = FVector::Dist(Self, S.Location);
		if (Dist < SoldierRange && Dist < BestDist)
		{
			Best = i;
			BestDist = Dist;
		}
	}
	for (int32 i = 0; i < Farmers.Num(); ++i)
	{
		const FCandidate& F = Farmers[i];
		if (!F.bValid || F.bDyingOrDead)
		{
			continue;
		}
		const float Dist = FVector::Dist(Self, F.Location);
		if (Dist < FarmerRange && Dist < BestDist)
		{
			Best = Soldiers.Num() + i;
			BestDist = Dist;
		}
	}
	return Best;
}

AActor* AZombieBase::PickAndSetTarget(const TArray<AActor*>& Soldiers, const TArray<AActor*>& Farmers)
{
	CD_SCOPE(STAT_CD_PickTarget);
	const UCDSettings* Settings = GetDefault<UCDSettings>();
	AActor* Player = UGameplayStatics::GetPlayerCharacter(this, 0);
	const FVector PlayerLoc = Player ? Player->GetActorLocation() : FVector::ZeroVector;

	TArray<FCandidate> S, F;
	S.Reserve(Soldiers.Num());
	F.Reserve(Farmers.Num());
	for (const AActor* A : Soldiers)
	{
		// Soldier "dying" = LifeSpan set (BP_Zombie: GetLifeSpan < 0.01 means alive).
		S.Add({ IsValid(A) ? A->GetActorLocation() : FVector::ZeroVector, IsValid(A), IsValid(A) && A->GetLifeSpan() >= 0.01f });
	}
	for (const AActor* A : Farmers)
	{
		F.Add({ IsValid(A) ? A->GetActorLocation() : FVector::ZeroVector, IsValid(A), ReadBlueprintBool(A, TEXT("Dead")) });
	}
	const int32 Pick = ChooseTarget(GetActorLocation(), Player ? &PlayerLoc : nullptr, S, F,
		Settings->ZombieSoldierRange, Settings->ZombieSoldierHeightRange, Settings->ZombieFarmerRange);
	AActor* Best = Pick == INDEX_NONE ? Player : (Pick < Soldiers.Num() ? Soldiers[Pick] : Farmers[Pick - Soldiers.Num()]);
	SetOwner(Best);
	return Best;
}

AActor* AZombieBase::PickAndSetTargetFromRegistry()
{
	UCrowdSubsystem* Crowd = UCrowdSubsystem::Get(this);
	return Crowd ? PickAndSetTarget(Crowd->GetSoldiers(), Crowd->GetFarmers()) : PickAndSetTarget({}, {});
}

int32 AZombieBase::GetThinkDivisor() const
{
	const UCDSettings* Settings = GetDefault<UCDSettings>();
	const float Dist = DistanceToPlayer();
	return Dist < Settings->ZombieNearPlayer ? 1 : (Dist < Settings->ZombieMidPlayer ? 2 : 4);
}

bool AZombieBase::IsFullDetail() const
{
	return DistanceToPlayer() < GetDefault<UCDSettings>()->ZombieNearPlayer || WasRecentlyRendered(0.2f);
}

TArray<AActor*> UCrowdSubsystem::Resolve(const TArray<TWeakObjectPtr<AActor>>& List)
{
	TArray<AActor*> Out;
	Out.Reserve(List.Num());
	for (const TWeakObjectPtr<AActor>& Weak : List)
	{
		if (AActor* Actor = Weak.Get())
		{
			Out.Add(Actor);
		}
	}
	return Out;
}

TArray<AActor*> UCrowdSubsystem::GetSoldiers() const
{
	return Resolve(Soldiers);
}

TArray<AActor*> UCrowdSubsystem::GetFarmers() const
{
	return Resolve(Farmers);
}

int32 UCrowdSubsystem::GetAliveZombieCount() const
{
	int32 Count = 0;
	for (const TWeakObjectPtr<AZombieBase>& Weak : Zombies)
	{
		const AZombieBase* Zombie = Weak.Get();
		if (Zombie && !AZombieBase::ReadBlueprintBool(Zombie, TEXT("Dead")))
		{
			++Count;
		}
	}
	return Count;
}

void UCrowdSubsystem::GetZombiesNear(FVector Location, float Radius, TArray<AActor*>& OutZombies) const
{
	CD_SCOPE(STAT_CD_ZombiesNear);
	OutZombies.Reset();
	const float RadiusSq = Radius * Radius;
	for (const TWeakObjectPtr<AZombieBase>& Weak : Zombies)
	{
		AZombieBase* Zombie = Weak.Get();
		if (Zombie && FVector::DistSquared(Zombie->GetActorLocation(), Location) <= RadiusSq && !AZombieBase::ReadBlueprintBool(Zombie, TEXT("Dead")))
		{
			OutZombies.Add(Zombie);
		}
	}
}

void UCrowdSubsystem::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
	CD_SCOPE(STAT_CD_CrowdTick);
	Zombies.RemoveAll([](const TWeakObjectPtr<AZombieBase>& Weak) { return !Weak.IsValid(); });
	Soldiers.RemoveAll([](const TWeakObjectPtr<AActor>& Weak) { return !Weak.IsValid(); });
	Farmers.RemoveAll([](const TWeakObjectPtr<AActor>& Weak) { return !Weak.IsValid(); });
	SET_DWORD_STAT(STAT_CD_Zombies, Zombies.Num());

	Accum += DeltaTime;
	if (Accum < UpdateInterval)
	{
		return;
	}
	Accum = 0.f;
	if (bManageBodyCollision)
	{
		ApplyBodyCollisionRule();
	}
}

void UCrowdSubsystem::ApplyBodyCollisionRule()
{
	// Same rule as BPC_CrowdTier (crowd/README_C1.md): body collision on screen, near the player or soldier-engaged (CustomTimeDilation 0.999); off elsewhere.
	const AActor* Player = UGameplayStatics::GetPlayerPawn(this, 0);
	const float RangeSq = FullDetailRange * FullDetailRange;
	for (const TWeakObjectPtr<AZombieBase>& Weak : Zombies)
	{
		AZombieBase* Zombie = Weak.Get();
		if (!Zombie || AZombieBase::ReadBlueprintBool(Zombie, TEXT("Dead")))
		{
			continue;
		}
		USkeletalMeshComponent* Mesh = Zombie->GetMesh();
		if (!Mesh)
		{
			continue;
		}
		const bool bNear = !Player || FVector::DistSquared(Zombie->GetActorLocation(), Player->GetActorLocation()) < RangeSq;
		const bool bFull = bNear || Zombie->WasRecentlyRendered(0.2f) || !FMath::IsNearlyEqual(Zombie->CustomTimeDilation, 1.f);
		Mesh->SetCollisionEnabled(bFull ? ECollisionEnabled::QueryOnly : ECollisionEnabled::NoCollision);
	}
}

TStatId UCrowdSubsystem::GetStatId() const
{
	RETURN_QUICK_DECLARE_CYCLE_STAT(UCrowdSubsystem, STATGROUP_Tickables);
}

// #500 point 5: a versioned save game. Every save carries SaveVersion; Migrate() upgrades old saves step by step, so changing the game
// never breaks a player's save. Data is kept in simple named maps, so new values need no new C++ (Blueprints: SetInt / GetInt ...).
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/SaveGame.h"
#include "Subsystems/GameInstanceSubsystem.h"
#include "CDSaveGame.generated.h"

UCLASS(BlueprintType)
class CASTLEDEFENDER_API UCDSaveGame : public USaveGame
{
	GENERATED_BODY()

public:
	// Raise when the meaning of saved data changes, and add a step in Migrate().
	static constexpr int32 CurrentVersion = 1;

	UPROPERTY(SaveGame, BlueprintReadOnly, Category = "Save")
	int32 SaveVersion = CurrentVersion;

	UPROPERTY(SaveGame, BlueprintReadOnly, Category = "Save")
	FDateTime SavedAt;

	UPROPERTY(SaveGame) TMap<FName, int32> Ints;
	UPROPERTY(SaveGame) TMap<FName, float> Floats;
	UPROPERTY(SaveGame) TMap<FName, FString> Strings;
	UPROPERTY(SaveGame) TSet<FName> Flags;

	UFUNCTION(BlueprintCallable, Category = "Save") void SetInt(FName Key, int32 Value) { Ints.Add(Key, Value); }
	UFUNCTION(BlueprintPure, Category = "Save") int32 GetInt(FName Key, int32 Default = 0) const { const int32* V = Ints.Find(Key); return V ? *V : Default; }
	UFUNCTION(BlueprintCallable, Category = "Save") void SetFloat(FName Key, float Value) { Floats.Add(Key, Value); }
	UFUNCTION(BlueprintPure, Category = "Save") float GetFloat(FName Key, float Default = 0.f) const { const float* V = Floats.Find(Key); return V ? *V : Default; }
	UFUNCTION(BlueprintCallable, Category = "Save") void SetString(FName Key, const FString& Value) { Strings.Add(Key, Value); }
	UFUNCTION(BlueprintPure, Category = "Save") FString GetString(FName Key) const { const FString* V = Strings.Find(Key); return V ? *V : FString(); }
	UFUNCTION(BlueprintCallable, Category = "Save") void SetFlag(FName Key, bool bOn) { if (bOn) { Flags.Add(Key); } else { Flags.Remove(Key); } }
	UFUNCTION(BlueprintPure, Category = "Save") bool HasFlag(FName Key) const { return Flags.Contains(Key); }

	// Upgrades an older save to CurrentVersion. Returns false if the save is newer than this game build (do not overwrite it).
	bool Migrate();
};

// Save / load with a safety copy: before overwriting slot X it copies X to X_bak, so one bad save never loses the player's progress.

{
	GENERATED_BODY()

public:
	UFUNCTION(BlueprintCallable, Category = "Save")

#include "CDSaveGame.h"
#include "Kismet/GameplayStatics.h"

bool UCDSaveGame::Migrate()
{
	if (SaveVersion > CurrentVersion)
	{
		return false;
	}
	// Add one step per version bump, oldest first. Example for a future version 2:
	// if (SaveVersion < 2) { Ints.Add(TEXT("Gems"), 0); SaveVersion = 2; }
	SaveVersion = CurrentVersion;
	return true;
}

UCDSaveGame* UCDSaveSubsystem::LoadOrCreate(const FString& Slot)
{
	UCDSaveGame* Data = nullptr;
	if (UGameplayStatics::DoesSaveGameExist(Slot, 0))
	{
		Data = Cast<UCDSaveGame>(UGameplayStatics::LoadGameFromSlot(Slot, 0));
	}
	if (!Data && UGameplayStatics::DoesSaveGameExist(Slot + TEXT("_bak"), 0))
	{
		UE_LOG(LogTemp, Warning, TEXT("CDSave: %s unreadable, using %s_bak"), *Slot, *Slot);
		Data = Cast<UCDSaveGame>(UGameplayStatics::LoadGameFromSlot(Slot + TEXT("_bak"), 0));
	}
	if (Data && !Data->Migrate())
	{
		UE_LOG(LogTemp, Error, TEXT("CDSave: %s is version %d, newer than this build (%d); loaded read-only"), *Slot, Data->SaveVersion, UCDSaveGame::CurrentVersion);
	}
	if (!Data)
	{
		Data = Cast<UCDSaveGame>(UGameplayStatics::CreateSaveGameObject(UCDSaveGame::StaticClass()));
	}
	Current = Data;
	return Data;
}

bool UCDSaveSubsystem::Save(UCDSaveGame* Data, const FString& Slot)
{
	if (!Data)
	{
		return false;
	}
	if (Data->SaveVersion > UCDSaveGame::CurrentVersion)
	{
		UE_LOG(LogTemp, Error, TEXT("CDSave: refusing to overwrite %s with a newer-version save"), *Slot);
		return false;
	}
	if (UGameplayStatics::DoesSaveGameExist(Slot, 0))
	{
		if (USaveGame* Old = UGameplayStatics::LoadGameFromSlot(Slot, 0))
		{
			UGameplayStatics::SaveGameToSlot(Old, Slot + TEXT("_bak"), 0);
		}
	}
	Data->SaveVersion = UCDSaveGame::CurrentVersion;
	Data->SavedAt = FDateTime::UtcNow();
	Current = Data;
	return UGameplayStatics::SaveGameToSlot(Data, Slot, 0);
}

#include "CDPoolSubsystem.h"
#include "CDStats.h"
#include "Engine/World.h"
#include "TimerManager.h"

DECLARE_CYCLE_STAT(TEXT("Pool Acquire"), STAT_CD_PoolAcquire, STATGROUP_CastleDefender);

void ICDPoolable::OnAcquired_Implementation() {}
void ICDPoolable::OnReleased_Implementation() {}

bool UCDPoolSubsystem::DoesSupportWorldType(const EWorldType::Type WorldType) const
{
	return WorldType == EWorldType::Game || WorldType == EWorldType::PIE;
}

AActor* UCDPoolSubsystem::AcquirePooledActor(TSubclassOf<AActor> Class, const FTransform& Transform)
{
	CD_SCOPE(STAT_CD_PoolAcquire);
	UWorld* World = GetWorld();
	if (!World || !*Class)
	{
		return nullptr;
	}
	AActor* Actor = nullptr;
	if (TArray<TWeakObjectPtr<AActor>>* List = Free.Find(*Class))
	{
		while (List->Num() > 0 && !Actor)
		{
			Actor = List->Pop().Get();
		}
	}
	if (Actor)
	{
		Actor->SetActorTransform(Transform, false, nullptr, ETeleportType::ResetPhysics);
		Actor->SetActorHiddenInGame(false);
		Actor->SetActorEnableCollision(true);
		Actor->SetActorTickEnabled(true);
	}
	else
	{
		FActorSpawnParameters Params;
		Params.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AlwaysSpawn;
		Actor = World->SpawnActor<AActor>(Class, Transform, Params);
	}
	if (Actor && Actor->GetClass()->ImplementsInterface(UCDPoolable::StaticClass()))
	{
		ICDPoolable::Execute_OnAcquired(Actor);
	}
	return Actor;
}

void UCDPoolSubsystem::ReleasePooledActor(AActor* Actor, float LifeSeconds)
{
	if (!IsValid(Actor))
	{
		return;
	}
	if (LifeSeconds > 0.f)
	{
		FTimerHandle Handle;
		TWeakObjectPtr<AActor> Weak(Actor);
		GetWorld()->GetTimerManager().SetTimer(Handle, FTimerDelegate::CreateWeakLambda(this, [this, Weak]()
		{
			if (AActor* A = Weak.Get()) { Park(A); }
		}), LifeSeconds, false);
		return;
	}
	Park(Actor);
}

void UCDPoolSubsystem::Park(AActor* Actor)
{
	if (Actor->GetClass()->ImplementsInterface(UCDPoolable::StaticClass()))
	{
		ICDPoolable::Execute_OnReleased(Actor);
	}
	TArray<TWeakObjectPtr<AActor>>& List = Free.FindOrAdd(Actor->GetClass());
	if (List.Num() >= MaxPerClass)
	{
		Actor->Destroy();
		return;
	}
	Actor->SetActorHiddenInGame(true);
	Actor->SetActorEnableCollision(false);
	Actor->SetActorTickEnabled(false);
	List.Add(Actor);
}

int32 UCDPoolSubsystem::GetPooledCount(TSubclassOf<AActor> Class) const
{
	const TArray<TWeakObjectPtr<AActor>>* List = Free.Find(*Class);
	return List ? List->Num() : 0;
}

#include "CDEditorTools.h"
#include "AssetRegistry/AssetRegistryModule.h"
#include "EdGraph/EdGraph.h"
#include "Engine/Blueprint.h"
#include "K2Node_CallFunction.h"

namespace
{
	TArray<UBlueprint*> LoadBlueprints(const FString& Path)
	{
		FAssetRegistryModule& Registry = FModuleManager::LoadModuleChecked<FAssetRegistryModule>(TEXT("AssetRegistry"));
		FARFilter Filter;
		Filter.PackagePaths.Add(FName(*Path));
		Filter.bRecursivePaths = true;
		Filter.ClassPaths.Add(UBlueprint::StaticClass()->GetClassPathName());
		Filter.bRecursiveClasses = true;
		TArray<FAssetData> Assets;
		Registry.Get().GetAssets(Filter, Assets);
		TArray<UBlueprint*> Out;
		for (const FAssetData& Asset : Assets)
		{
			if (UBlueprint* BP = Cast<UBlueprint>(Asset.GetAsset()))
			{
				Out.Add(BP);
			}
		}
		return Out;
	}

	TArray<UEdGraph*> AllGraphs(UBlueprint* BP)
	{
		TArray<UEdGraph*> Graphs;
		BP->GetAllGraphs(Graphs);
		return Graphs;
	}
}

TArray<FString> UCDEditorTools::FindBlueprintsWithErrors(const FString& Path)
{
	TArray<FString> Lines;
	for (UBlueprint* BP : LoadBlueprints(Path))
	{
		bool bNodeError = false;
		for (UEdGraph* Graph : AllGraphs(BP))
		{
			for (const UEdGraphNode* Node : Graph->Nodes)
			{
				bNodeError |= Node && Node->bHasCompilerMessage && Node->ErrorType <= EMessageSeverity::Error;
			}
		}
		if (BP->Status == BS_Error || bNodeError)
		{
			Lines.Add(FString::Printf(TEXT("%s | status %d | node errors %s"), *BP->GetPathName(), (int32)BP->Status, bNodeError ? TEXT("yes") : TEXT("no")));
		}
	}
	for (const FString& L : Lines) { UE_LOG(LogTemp, Warning, TEXT("CD errors: %s"), *L); }
	return Lines;
}

TArray<FString> UCDEditorTools::FindExpensiveCalls(const FString& Path)
{
	static const TSet<FName> Expensive = {
		TEXT("GetAllActorsOfClass"), TEXT("GetAllActorsWithTag"), TEXT("GetAllActorsOfClassWithTag"), TEXT("GetActorOfClass"),
		TEXT("GetAllActorsWithInterface"), TEXT("GetAllWidgetsOfClass"), TEXT("LineOfSightTo"), TEXT("GetComponentsByClass"),
		TEXT("LineTraceSingle"), TEXT("LineTraceMulti"), TEXT("SphereOverlapActors"), TEXT("FindNearestActor") };
	TArray<FString> Lines;
	for (UBlueprint* BP : LoadBlueprints(Path))
	{
		for (UEdGraph* Graph : AllGraphs(BP))
		{
			for (UEdGraphNode* Node : Graph->Nodes)
			{
				const UK2Node_CallFunction* Call = Cast<UK2Node_CallFunction>(Node);
				if (Call && Expensive.Contains(Call->FunctionReference.GetMemberName()))
				{
					Lines.Add(FString::Printf(TEXT("%s | %s | %s | %s"), *BP->GetName(), *Graph->GetName(), *Node->GetName(), *Call->FunctionReference.GetMemberName().ToString()));
				}
			}
		}
	}
	for (const FString& L : Lines) { UE_LOG(LogTemp, Display, TEXT("CD expensive: %s"), *L); }
	return Lines;
}
