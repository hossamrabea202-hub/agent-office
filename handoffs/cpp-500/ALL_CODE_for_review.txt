// #500: C++ parent for BP_Zombie. Holds the per-Think hot loops natively; BP_Zombie keeps its look, attacks, montages and states.
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Character.h"
#include "ZombieBase.generated.h"

UCLASS(Abstract, Blueprintable)
class CASTLEDEFENDER_API AZombieBase : public ACharacter
{
	GENERATED_BODY()

public:
	AZombieBase();

	// BP_Zombie Think's target pick (nodes CX_501..zfTargetFarmer17), in one native call. Same rules as the live graph:
	// start from the player; a soldier that is not dying (LifeSpan < 0.01), within ZombieSoldierHeightRange height and ZombieSoldierRange, if strictly nearer;
	// then a farmer that is not Dead, within ZombieFarmerRange, if strictly nearer (equal distance: the earlier pick wins, soldiers before farmers).
	// Writes CurrentTarget. While BP_Zombie still reads GetOwner() as its target, it also sets Owner (bAlsoSetOwner); drop that once the Blueprint reads CurrentTarget.
	UFUNCTION(BlueprintCallable, Category = "Zombie|Brain")
	AActor* PickAndSetTarget(const TArray<AActor*>& Soldiers, const TArray<AActor*>& Farmers);

	// The same, reading soldiers and farmers straight from the crowd registry (no arrays built per call).
	UFUNCTION(BlueprintCallable, Category = "Zombie|Brain")
	AActor* PickAndSetTargetFromRegistry();

	// The current target, or null if it was destroyed since the pick (weak reference: a dead target is never kept alive).
	UFUNCTION(BlueprintPure, Category = "Zombie|Brain")
	AActor* GetCurrentTarget() const { AActor* T = CurrentTarget.Get(); return IsValid(T) ? T : nullptr; }

	// Call at the START of the death transition (before the death code sets its collision): the crowd manager stops touching this body.
	UFUNCTION(BlueprintCallable, Category = "Zombie|Crowd")
	void NotifyDying();

	// Think's "heavy part every Nth tick" divisor: 1 within ZombieNearPlayer, 2 within ZombieMidPlayer, else 4 (K2Node_IfThenElse_11).
	UFUNCTION(BlueprintPure, Category = "Zombie|Brain")
	int32 GetThinkDivisor() const;

	// Near the player (ZombieNearPlayer) or rendered in the last 0.2 s (K2Node_CallFunction_42 / _43).
	UFUNCTION(BlueprintPure, Category = "Zombie|Brain")
	bool IsFullDetail() const;

	// Dead for the crowd's purposes: being destroyed, or NotifyDying was called, or the Blueprint bool "Dead" is true. Override when a C++ death state exists.
	virtual bool IsDeadForCrowd() const;

	// Engaged by a soldier. Today BP_SoldierSword marks its zombie with CustomTimeDilation 0.999 (fragile; replace with an explicit state later).
	virtual bool IsEngagedForCrowd() const;

	// Reads a Blueprint bool variable from any object. bFound tells whether the variable exists.
	static bool ReadBlueprintBool(const UObject* Object, FName Name, bool* bFound = nullptr);

	// The pure target rule (no world), shared by PickAndSetTarget and the unit tests. Feed candidates with Consider(); Best is the chosen index or INDEX_NONE.
	struct FTargetRule
	{
		FVector Self;
		float BestDistSq;
		float SoldierRangeSq, SoldierHeightRange, FarmerRangeSq;
		int32 Best = INDEX_NONE;

		FTargetRule(const FVector& InSelf, const FVector* Player, float SoldierRange, float InSoldierHeightRange, float FarmerRange, bool bNoPlayerBlocks);
		void ConsiderSoldier(int32 Index, const FVector& Location, bool bValid, bool bDying);
		void ConsiderFarmer(int32 Index, const FVector& Location, bool bValid, bool bDead);
	};

	// Test helper: candidates as plain data. Returns the chosen index in Soldiers (0..), Farmers (Soldiers.Num()..), or INDEX_NONE (keep the player).
	struct FCandidate
	{
		FVector Location;
		bool bValid = true;
		bool bDyingOrDead = false;
	};
	static int32 ChooseTarget(const FVector& Self, const FVector* Player, const TArray<FCandidate>& Soldiers, const TArray<FCandidate>& Farmers,
		float SoldierRange, float SoldierHeightRange, float FarmerRange, bool bNoPlayerBlocks = true);

protected:
	virtual void BeginPlay() override;
	virtual void EndPlay(const EEndPlayReason::Type EndPlayReason) override;

	TWeakObjectPtr<AActor> CurrentTarget;
	bool bDying = false;

	// Transition: BP_Zombie still uses Owner as its target slot. Turn off when the Blueprint reads CurrentTarget (Owner also drives net ownership).
	UPROPERTY(EditDefaultsOnly, Category = "Zombie|Brain")
	bool bAlsoSetOwner = true;

private:
	float DistanceToPlayer() const;
	AActor* FinishPick(AActor* Player, AActor* Picked);
};

#include "ZombieBase.h"
#include "CDSettings.h"
#include "CDStats.h"
#include "CrowdSubsystem.h"
#include "Kismet/GameplayStatics.h"
#include "UObject/UnrealType.h"

DECLARE_CYCLE_STAT(TEXT("Zombie PickTarget"), STAT_CD_PickTarget, STATGROUP_CastleDefender);

AZombieBase::AZombieBase()
{
	PrimaryActorTick.bCanEverTick = true;
}

void AZombieBase::BeginPlay()
{
	Super::BeginPlay();   // runs the Blueprint's Event BeginPlay; registration below does not depend on it
	if (UCrowdSubsystem* Crowd = UCrowdSubsystem::Get(this))
	{
		Crowd->RegisterZombie(this);
	}
}

void AZombieBase::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	if (UCrowdSubsystem* Crowd = UCrowdSubsystem::Get(this))
	{
		Crowd->UnregisterZombie(this);
	}
	CurrentTarget = nullptr;
	Super::EndPlay(EndPlayReason);
}

bool AZombieBase::ReadBlueprintBool(const UObject* Object, FName Name, bool* bFound)
{
	if (bFound) { *bFound = false; }
	if (!IsValid(Object))
	{
		return false;
	}
	const FBoolProperty* Prop = FindFProperty<FBoolProperty>(Object->GetClass(), Name);
	if (bFound) { *bFound = Prop != nullptr; }
	return Prop && Prop->GetPropertyValue_InContainer(Object);
}

bool AZombieBase::IsDeadForCrowd() const
{
	// Destruction and NotifyDying always win; the Blueprint flag can only add "dead", never hide it.
	return IsActorBeingDestroyed() || !IsValid(this) || bDying || ReadBlueprintBool(this, TEXT("Dead"));
}

void AZombieBase::NotifyDying()
{
	bDying = true;
	if (UCrowdSubsystem* Crowd = UCrowdSubsystem::Get(this))
	{
		Crowd->RelinquishBodyCollision(this);
	}
}

bool AZombieBase::IsEngagedForCrowd() const
{
	return !FMath::IsNearlyEqual(CustomTimeDilation, 1.f);
}

float AZombieBase::DistanceToPlayer() const
{
	const AActor* Player = UGameplayStatics::GetPlayerCharacter(this, 0);
	// GetDistanceTo(None) is 0 in Blueprint: keep that, so a missing player counts as near.
	return Player ? GetDistanceTo(Player) : 0.f;
}

AZombieBase::FTargetRule::FTargetRule(const FVector& InSelf, const FVector* Player, float SoldierRange, float InSoldierHeightRange, float FarmerRange, bool bNoPlayerBlocks)
	: Self(InSelf)
	, BestDistSq(0.f)
	, SoldierRangeSq(FMath::Square(FMath::Max(0.f, SoldierRange)))       // settings from .ini are not clamped by the editor UI
	, SoldierHeightRange(FMath::Max(0.f, InSoldierHeightRange))
	, FarmerRangeSq(FMath::Square(FMath::Max(0.f, FarmerRange)))
{
	// Blueprint parity: with no player, GetDistanceTo(None) = 0, so nothing is "nearer" and no soldier / farmer is picked.
	// bNoPlayerBlocks = false is the alternative (pick the nearest valid target); TGA's SPEC_zombie.md decides which is intended.
	BestDistSq = Player ? FVector::DistSquared(Self, *Player) : (bNoPlayerBlocks ? 0.f : TNumericLimits<float>::Max());
}

void AZombieBase::FTargetRule::ConsiderSoldier(int32 Index, const FVector& Location, bool bValid, bool bDying)
{
	if (!bValid || bDying || FMath::Abs(Location.Z - Self.Z) >= SoldierHeightRange)
	{
		return;
	}
	const float DistSq = FVector::DistSquared(Self, Location);
	if (DistSq < SoldierRangeSq && DistSq < BestDistSq)
	{
		Best = Index;
		BestDistSq = DistSq;
	}
}

void AZombieBase::FTargetRule::ConsiderFarmer(int32 Index, const FVector& Location, bool bValid, bool bDead)
{
	if (!bValid || bDead)
	{
		return;
	}
	const float DistSq = FVector::DistSquared(Self, Location);
	if (DistSq < FarmerRangeSq && DistSq < BestDistSq)
	{
		Best = Index;
		BestDistSq = DistSq;
	}
}

int32 AZombieBase::ChooseTarget(const FVector& Self, const FVector* Player, const TArray<FCandidate>& Soldiers, const TArray<FCandidate>& Farmers,
	float SoldierRange, float SoldierHeightRange, float FarmerRange, bool bNoPlayerBlocks)
{
	FTargetRule Rule(Self, Player, SoldierRange, SoldierHeightRange, FarmerRange, bNoPlayerBlocks);
	for (int32 i = 0; i < Soldiers.Num(); ++i)
	{
		Rule.ConsiderSoldier(i, Soldiers[i].Location, Soldiers[i].bValid, Soldiers[i].bDyingOrDead);
	}
	for (int32 i = 0; i < Farmers.Num(); ++i)
	{
		Rule.ConsiderFarmer(Soldiers.Num() + i, Farmers[i].Location, Farmers[i].bValid, Farmers[i].bDyingOrDead);
	}
	return Rule.Best;
}

AActor* AZombieBase::FinishPick(AActor* Player, AActor* Picked)
{
	AActor* Best = Picked ? Picked : Player;
	CurrentTarget = Best;
	if (bAlsoSetOwner)
	{
		SetOwner(Best);
	}
	return Best;
}

// "Dying" for a soldier = a life span is set (Blueprint parity: GetLifeSpan < 0.01 means alive). Replace with an explicit state with ASoldierBase.
static bool IsSoldierDying(const AActor* Soldier) { return Soldier->GetLifeSpan() >= 0.01f; }

AActor* AZombieBase::PickAndSetTarget(const TArray<AActor*>& Soldiers, const TArray<AActor*>& Farmers)
{
	CD_SCOPE(STAT_CD_PickTarget);
	const UCDSettings* S = GetDefault<UCDSettings>();
	AActor* Player = UGameplayStatics::GetPlayerCharacter(this, 0);
	const FVector PlayerLoc = Player ? Player->GetActorLocation() : FVector::ZeroVector;
	FTargetRule Rule(GetActorLocation(), Player ? &PlayerLoc : nullptr, S->ZombieSoldierRange, S->ZombieSoldierHeightRange, S->ZombieFarmerRange, S->bZombieNoPlayerBlocksTargets);
	for (int32 i = 0; i < Soldiers.Num(); ++i)
	{
		const AActor* A = Soldiers[i];
		if (IsValid(A)) { Rule.ConsiderSoldier(i, A->GetActorLocation(), true, IsSoldierDying(A)); }
	}
	for (int32 i = 0; i < Farmers.Num(); ++i)
	{
		const AActor* A = Farmers[i];
		if (IsValid(A)) { Rule.ConsiderFarmer(Soldiers.Num() + i, A->GetActorLocation(), true, ReadBlueprintBool(A, TEXT("Dead"))); }
	}
	AActor* Picked = Rule.Best == INDEX_NONE ? nullptr : (Rule.Best < Soldiers.Num() ? Soldiers[Rule.Best] : Farmers[Rule.Best - Soldiers.Num()]);
	return FinishPick(Player, Picked);
}

AActor* AZombieBase::PickAndSetTargetFromRegistry()
{
	CD_SCOPE(STAT_CD_PickTarget);
	const UCDSettings* S = GetDefault<UCDSettings>();
	AActor* Player = UGameplayStatics::GetPlayerCharacter(this, 0);
	const UCrowdSubsystem* Crowd = UCrowdSubsystem::Get(this);
	if (!Crowd)
	{
		return FinishPick(Player, nullptr);
	}
	const FVector PlayerLoc = Player ? Player->GetActorLocation() : FVector::ZeroVector;
	FTargetRule Rule(GetActorLocation(), Player ? &PlayerLoc : nullptr, S->ZombieSoldierRange, S->ZombieSoldierHeightRange, S->ZombieFarmerRange, S->bZombieNoPlayerBlocksTargets);
	AActor* Picked = nullptr;
	int32 Index = 0;
	Crowd->ForEachSoldier([&](AActor* A)
	{
		const int32 Before = Rule.Best;
		Rule.ConsiderSoldier(Index++, A->GetActorLocation(), true, IsSoldierDying(A));
		if (Rule.Best != Before) { Picked = A; }
	});
	Crowd->ForEachFarmer([&](AActor* A)
	{
		const int32 Before = Rule.Best;
		Rule.ConsiderFarmer(Index++, A->GetActorLocation(), true, ReadBlueprintBool(A, TEXT("Dead")));
		if (Rule.Best != Before) { Picked = A; }
	});
	return FinishPick(Player, Picked);
}

int32 AZombieBase::GetThinkDivisor() const
{
	const UCDSettings* S = GetDefault<UCDSettings>();
	const float Dist = DistanceToPlayer();
	return Dist < S->ZombieNearPlayer ? 1 : (Dist < S->ZombieMidPlayer ? 2 : 4);
}

bool AZombieBase::IsFullDetail() const
{
	return DistanceToPlayer() < GetDefault<UCDSettings>()->ZombieNearPlayer || WasRecentlyRendered(0.2f);
}

// #500: one native registry and manager for the battle (point 4, shared systems): zombies, soldiers and farmers register here once,
// so no Blueprint needs GetAllActorsOfClass / GetAllActorsWithTag to find them again.
// - zombies register themselves (AZombieBase, in its native BeginPlay / EndPlay: Blueprint children need no "call parent" for this);
// - soldiers and farmers call RegisterSoldier / RegisterFarmer from their BeginPlay and EndPlay until they get C++ parents;
// - optional: the C1 body-collision rule for every zombie in one loop (off by default while BPC_CrowdTier still does it).
//   It remembers each zombie's own collision setting and puts it back when the zombie comes into full detail again, unregisters ALIVE, or the rule is
//   switched off. The moment a zombie is dying or dead it lets go WITHOUT restoring anything: from then on the death code owns the body collision
//   (call AZombieBase::NotifyDying at the start of the death transition, before the death code sets its collision; the crowd also detects Dead itself);
// - native radius queries (the C3 job, without a sphere sweep);
// - `cd.CrowdReport` (console): registry counts against the actors really in the world, to catch a spawn path that does not register.
#pragma once

#include "CoreMinimal.h"
#include "Engine/EngineTypes.h"
#include "Subsystems/WorldSubsystem.h"
#include "CrowdSubsystem.generated.h"

class AZombieBase;

UCLASS()
class CASTLEDEFENDER_API UCrowdSubsystem : public UTickableWorldSubsystem
{
	GENERATED_BODY()

public:
	static UCrowdSubsystem* Get(const UObject* WorldContext);

	void RegisterZombie(AZombieBase* Zombie);
	void UnregisterZombie(AZombieBase* Zombie);

	// Death transition starts: stop managing this zombie's body collision and forget its stored setting (nothing is restored).
	void RelinquishBodyCollision(AZombieBase* Zombie);

	// Soldiers / farmers are still pure Blueprints: call these from their BeginPlay and EndPlay (every spawn path).
	UFUNCTION(BlueprintCallable, Category = "Crowd", meta = (DefaultToSelf = "Soldier"))
	void RegisterSoldier(AActor* Soldier);

	UFUNCTION(BlueprintCallable, Category = "Crowd", meta = (DefaultToSelf = "Soldier"))
	void UnregisterSoldier(AActor* Soldier);

	UFUNCTION(BlueprintCallable, Category = "Crowd", meta = (DefaultToSelf = "Farmer"))
	void RegisterFarmer(AActor* Farmer);

	UFUNCTION(BlueprintCallable, Category = "Crowd", meta = (DefaultToSelf = "Farmer"))
	void UnregisterFarmer(AActor* Farmer);

	// Copies for Blueprint. C++ should use ForEachSoldier / ForEachFarmer (no array per call).
	UFUNCTION(BlueprintCallable, Category = "Crowd")
	TArray<AActor*> GetSoldiers() const;

	UFUNCTION(BlueprintCallable, Category = "Crowd")
	TArray<AActor*> GetFarmers() const;

	template <typename F> void ForEachSoldier(F&& Fn) const { for (const TWeakObjectPtr<AActor>& W : Soldiers) { if (AActor* A = W.Get()) { Fn(A); } } }
	template <typename F> void ForEachFarmer(F&& Fn) const { for (const TWeakObjectPtr<AActor>& W : Farmers) { if (AActor* A = W.Get()) { Fn(A); } } }
	// Every registered zombie, dead or alive (same set GetAllActorsOfClass would return for AZombieBase classes).
	template <typename F> void ForEachZombie(F&& Fn) const { for (const TWeakObjectPtr<AZombieBase>& W : Zombies) { if (AZombieBase* Z = W.Get()) { Fn(Z); } } }

	// Registered zombies that are not dead (see AZombieBase::IsDeadForCrowd).
	UFUNCTION(BlueprintPure, Category = "Crowd")
	int32 GetAliveZombieCount() const;

	// False when MaxAliveZombies (Project Settings > Castle Defender) is reached. Spawners call this before SpawnActor.
	UFUNCTION(BlueprintPure, Category = "Crowd")
	bool CanSpawnZombie() const;

	// Zombies (registered, not dead) within Radius of Location.
	UFUNCTION(BlueprintCallable, Category = "Crowd")
	void GetZombiesNear(FVector Location, float Radius, TArray<AActor*>& OutZombies) const;

	// When true, the C1 rule runs here for every zombie; then BPC_CrowdTier can be removed from BP_Zombie. Switching it off restores every zombie.
	UFUNCTION(BlueprintCallable, Category = "Crowd")
	void SetManageBodyCollision(bool bOn);

	UFUNCTION(BlueprintPure, Category = "Crowd")
	bool GetManageBodyCollision() const { return bManageBodyCollision; }

	void Report() const;

	virtual void Initialize(FSubsystemCollectionBase& Collection) override;
	virtual void Deinitialize() override;
	virtual void Tick(float DeltaTime) override;
	virtual TStatId GetStatId() const override;

protected:

	TArray<TWeakObjectPtr<AZombieBase>> Zombies;

void UCrowdSubsystem::GetZombiesNear(FVector Location, float Radius, TArray<AActor*>& OutZombies) const
{
	CD_SCOPE(STAT_CD_ZombiesNear);
	OutZombies.Reset();
	if (!(Radius > 0.f))
	{
		return;   // negative, zero or NaN radius: nothing
	}
	const float RadiusSq = Radius * Radius;
	for (const TWeakObjectPtr<AZombieBase>& Weak : Zombies)
	{
		AZombieBase* Zombie = Weak.Get();
		if (Zombie && FVector::DistSquared(Zombie->GetActorLocation(), Location) <= RadiusSq && !Zombie->IsDeadForCrowd())
		{
			OutZombies.Add(Zombie);
		}
	}
}

void UCrowdSubsystem::SetManageBodyCollision(bool bOn)

	if (bManageBodyCollision && !bOn)
	{
		RestoreAll();
	}
	bManageBodyCollision = bOn;
}

void UCrowdSubsystem::PurgeInvalid()
{
	Zombies.RemoveAll([](const TWeakObjectPtr<AZombieBase>& Weak) { return !Weak.IsValid(); });
	Soldiers.RemoveAll([](const TWeakObjectPtr<AActor>& Weak) { return !Weak.IsValid(); });
	Farmers.RemoveAll([](const TWeakObjectPtr<AActor>& Weak) { return !Weak.IsValid(); });
	for (auto It = Relinquished.CreateIterator(); It; ++It)
	{
		if (!It->IsValid())
		{
			It.RemoveCurrent();
		}
	}
	for (auto It = OriginalCollision.CreateIterator(); It; ++It)
	{
		if (!It.Key().IsValid())
		{
			It.RemoveCurrent();
		}
	}
}

void UCrowdSubsystem::Tick(float DeltaTime)
{
	// No Super::Tick: the base has nothing to do and the reviewer flagged it for 5.8.
	CD_SCOPE(STAT_CD_CrowdTick);
	Accum += DeltaTime;
	if (Accum < UpdateInterval)
	{
		return;
	}
	Accum = 0.f;
	PurgeInvalid();
	SET_DWORD_STAT(STAT_CD_Zombies, Zombies.Num());
	SET_DWORD_STAT(STAT_CD_Soldiers, Soldiers.Num());
	SET_DWORD_STAT(STAT_CD_Farmers, Farmers.Num());
	if (bManageBodyCollision)
	{
		ApplyBodyCollisionRule();
	}
}

void UCrowdSubsystem::RestoreBodyCollision(AZombieBase* Zombie)
{
	if (!Zombie)
	{
		return;
	}
	if (const TEnumAsByte<ECollisionEnabled::Type>* Original = OriginalCollision.Find(Zombie))
	{
		// Alive zombies only (callers check); only undo our own change (the mesh is still at the NoCollision we set).
		USkeletalMeshComponent* Mesh = Zombie->GetMesh();
		if (Mesh && !Mesh->IsSimulatingPhysics() && Mesh->GetCollisionEnabled() == ECollisionEnabled::NoCollision)
		{
			Mesh->SetCollisionEnabled(Original->GetValue());
		}
		OriginalCollision.Remove(Zombie);
	}
}

void UCrowdSubsystem::RestoreAll()
{
	for (const TWeakObjectPtr<AZombieBase>& Weak : Zombies)
	{
		AZombieBase* Zombie = Weak.Get();
		if (Zombie && !Zombie->IsDeadForCrowd() && !Relinquished.Contains(Zombie))
		{
			RestoreBodyCollision(Zombie);
		}
	}
	OriginalCollision.Reset();
	Relinquished.Reset();
}

void UCrowdSubsystem::ApplyBodyCollisionRule()
{
	// Same rule as BPC_CrowdTier (crowd/README_C1.md): body collision as the zombie had it on screen, near the player or engaged; off elsewhere.
	const AActor* Player = UGameplayStatics::GetPlayerPawn(this, 0);
	const float RangeSq = FullDetailRange * FullDetailRange;
	for (const TWeakObjectPtr<AZombieBase>& Weak : Zombies)
	{
		AZombieBase* Zombie = Weak.Get();
		if (!Zombie)
		{
			continue;
		}
		USkeletalMeshComponent* Mesh = Zombie->GetMesh();
		if (!Mesh || Relinquished.Contains(Zombie))
		{
			continue;
		}
		if (Mesh->IsSimulatingPhysics() || Zombie->IsDeadForCrowd())
		{
			RelinquishBodyCollision(Zombie);   // dying / ragdoll: the death code owns the body from now on, nothing is restored
			continue;
		}
		const bool bNear = !Player || FVector::DistSquared(Zombie->GetActorLocation(), Player->GetActorLocation()) < RangeSq;
		const bool bFull = bNear || Zombie->WasRecentlyRendered(0.2f) || Zombie->IsEngagedForCrowd();
		if (bFull)
		{
			RestoreBodyCollision(Zombie);
		}
		else if (Mesh->GetCollisionEnabled() != ECollisionEnabled::NoCollision)
		{
			OriginalCollision.FindOrAdd(Zombie, Mesh->GetCollisionEnabled());
			Mesh->SetCollisionEnabled(ECollisionEnabled::NoCollision);
		}
	}
}

void UCrowdSubsystem::Report() const
{
	int32 WorldZombies = 0;
	for (TActorIterator<AZombieBase> It(GetWorld()); It; ++It)
	{
		++WorldZombies;
	}
	UE_LOG(LogTemp, Display, TEXT("cd.CrowdReport: zombies registered %d / in world %d (alive %d, cap %d) | soldiers %d | farmers %d | body collision managed %s, %d changed"),
		Zombies.Num(), WorldZombies, GetAliveZombieCount(), MaxAliveZombies, Soldiers.Num(), Farmers.Num(), bManageBodyCollision ? TEXT("yes") : TEXT("no"), OriginalCollision.Num());
}

{
	RETURN_QUICK_DECLARE_CYCLE_STAT(UCrowdSubsystem, STATGROUP_Tickables);
}

// #500 C++ round 2: native parent for the HUD chain. BP_ZombieHUD (the bottom of the 11-level chain, parent HUD) becomes a child of ACDHudBase,
// so every HUD level (Minimap, SkillTree, Barracks, Army, Controls, Queue, Hero, Commander, Armory, Notice) inherits these functions.
// The HUD's drawing, menus, shop and input stay in the Blueprints. Only the expensive world lookups go native and get cached:
//   GetActorOfClass       -> GetCachedActorOfClass     (kept until that actor is gone; a missing class is looked for again at most every MissingRetry s)
//   GetAllActorsOfClass   -> GetCachedActorsOfClass    (zombie classes: read live from the crowd registry; other classes: reused for MaxAge s)
//   GetAllActorsWithTag   -> GetCachedActorsWithTag    (reused for MaxAge s)
// Same pins as the engine nodes, so hud_swap_scans_ue.py can swap them in place. Semantics differ in one way only, on purpose: a result can be up to
// MaxAge old (a new actor shows up to MaxAge later; destroyed actors are always filtered out). Use the engine node where that matters.
// `stat CastleDefender` shows "HUD DrawHUD (whole chain)" and the cache hit / miss counters.

#include "GameFramework/HUD.h"
#include "CDHudBase.generated.h"

class CASTLEDEFENDER_API ACDHudBase : public AHUD
{
	GENERATED_BODY()

public:
	UFUNCTION(BlueprintCallable, Category = "Castle Defender|HUD", meta = (DeterminesOutputType = "ActorClass"))
	AActor* GetCachedActorOfClass(TSubclassOf<AActor> ActorClass);

	UFUNCTION(BlueprintCallable, Category = "Castle Defender|HUD", meta = (DeterminesOutputType = "ActorClass", DynamicOutputParam = "OutActors"))
	void GetCachedActorsOfClass(TSubclassOf<AActor> ActorClass, TArray<AActor*>& OutActors);

	UFUNCTION(BlueprintCallable, Category = "Castle Defender|HUD")
	void GetCachedActorsWithTag(FName Tag, TArray<AActor*>& OutActors);

	// Rendered in the last Tolerance seconds (cheap; replaces per-actor line-of-sight traces for "is it visible" checks).
	UFUNCTION(BlueprintPure, Category = "Castle Defender|HUD")
	static bool IsActorOnScreen(const AActor* Actor, float Tolerance = 0.1f);

	// Drop every cache now (for example after a level change or a big spawn the HUD must show at once).
	UFUNCTION(BlueprintCallable, Category = "Castle Defender|HUD")
	void InvalidateHudCaches();

	// How old a class / tag list may be before it is scanned again, seconds.
	UPROPERTY(EditDefaultsOnly, BlueprintReadWrite, Category = "Castle Defender|HUD", meta = (ClampMin = "0", UIMax = "2"))
	float MaxAge = 0.25f;

	// How often a class with no actor is looked for again, seconds.
	UPROPERTY(EditDefaultsOnly, BlueprintReadWrite, Category = "Castle Defender|HUD", meta = (ClampMin = "0", UIMax = "5"))
	float MissingRetry = 0.5f;

	virtual void DrawHUD() override;

private:
	struct FListCache
	{
		TArray<TWeakObjectPtr<AActor>> Actors;
		double Time = -1.0;
	};
	struct FSingleCache
	{
		TWeakObjectPtr<AActor> Actor;
		double MissedAt = -1.0;
	};

	double Now() const;
	static void Resolve(const TArray<TWeakObjectPtr<AActor>>& In, TArray<AActor*>& Out);

	TMap<TWeakObjectPtr<UClass>, FSingleCache> Singles;
	TMap<TWeakObjectPtr<UClass>, FListCache> ClassLists;
	TMap<FName, FListCache> TagLists;

DECLARE_CYCLE_STAT(TEXT("HUD DrawHUD (whole chain)"), STAT_CD_HudDraw, STATGROUP_CastleDefender);
DECLARE_DWORD_ACCUMULATOR_STAT(TEXT("HUD cache hits"), STAT_CD_HudHits, STATGROUP_CastleDefender);
DECLARE_DWORD_ACCUMULATOR_STAT(TEXT("HUD cache scans"), STAT_CD_HudScans, STATGROUP_CastleDefender);

void ACDHudBase::DrawHUD()
{
	CD_SCOPE(STAT_CD_HudDraw);
	Super::DrawHUD();   // runs every Blueprint level's Receive Draw HUD
}

double ACDHudBase::Now() const
{
	const UWorld* World = GetWorld();
	return World ? World->GetTimeSeconds() : 0.0;
}

void ACDHudBase::Resolve(const TArray<TWeakObjectPtr<AActor>>& In, TArray<AActor*>& Out)
{
	Out.Reset(In.Num());
	for (const TWeakObjectPtr<AActor>& W : In)
	{
		AActor* A = W.Get();
		if (IsValid(A) && !A->IsActorBeingDestroyed())
		{
			Out.Add(A);
		}
	}
}

AActor* ACDHudBase::GetCachedActorOfClass(TSubclassOf<AActor> ActorClass)
{
	if (!*ActorClass)
	{
		return nullptr;
	}
	FSingleCache& C = Singles.FindOrAdd(*ActorClass);
	AActor* Cached = C.Actor.Get();
	if (IsValid(Cached) && !Cached->IsActorBeingDestroyed())
	{
		INC_DWORD_STAT(STAT_CD_HudHits);
		return Cached;
	}
	const double T = Now();
	if (C.MissedAt >= 0.0 && T - C.MissedAt < MissingRetry)
	{
		INC_DWORD_STAT(STAT_CD_HudHits);
		return nullptr;   // looked very recently and there was none
	}
	INC_DWORD_STAT(STAT_CD_HudScans);
	AActor* Found = UGameplayStatics::GetActorOfClass(this, ActorClass);
	C.Actor = Found;
	C.MissedAt = Found ? -1.0 : T;
	return Found;
}

void ACDHudBase::GetCachedActorsOfClass(TSubclassOf<AActor> ActorClass, TArray<AActor*>& OutActors)
{
	OutActors.Reset();
	if (!*ActorClass)
	{
		return;
	}
	// Zombies: the crowd registry is exact and costs nothing, no cache needed. Dead zombies included, like GetAllActorsOfClass.
	if (ActorClass->IsChildOf(AZombieBase::StaticClass()))
	{
		if (const UCrowdSubsystem* Crowd = UCrowdSubsystem::Get(this))
		{
			UClass* Wanted = *ActorClass;
			Crowd->ForEachZombie([&OutActors, Wanted](AZombieBase* Z)
			{
				if (Z->IsA(Wanted) && !Z->IsActorBeingDestroyed()) { OutActors.Add(Z); }
			});
			INC_DWORD_STAT(STAT_CD_HudHits);
			return;
		}
	}
	FListCache& C = ClassLists.FindOrAdd(*ActorClass);
	const double T = Now();
	if (C.Time < 0.0 || T - C.Time >= MaxAge)
	{
		INC_DWORD_STAT(STAT_CD_HudScans);
		TArray<AActor*> Found;
		UGameplayStatics::GetAllActorsOfClass(this, ActorClass, Found);
		C.Actors.Reset(Found.Num());
		for (AActor* A : Found) { C.Actors.Add(A); }
		C.Time = T;
	}
	else
	{
		INC_DWORD_STAT(STAT_CD_HudHits);
	}
	Resolve(C.Actors, OutActors);
}

void ACDHudBase::GetCachedActorsWithTag(FName Tag, TArray<AActor*>& OutActors)
{
	OutActors.Reset();
	FListCache& C = TagLists.FindOrAdd(Tag);
	const double T = Now();
	if (C.Time < 0.0 || T - C.Time >= MaxAge)
	{
		INC_DWORD_STAT(STAT_CD_HudScans);
		TArray<AActor*> Found;
		UGameplayStatics::GetAllActorsWithTag(this, Tag, Found);
		C.Actors.Reset(Found.Num());
		for (AActor* A : Found) { C.Actors.Add(A); }
		C.Time = T;
	}
	else
	{
		INC_DWORD_STAT(STAT_CD_HudHits);
	}
	Resolve(C.Actors, OutActors);
	// A tagged actor whose tag was removed since the scan is dropped here, so the list never shows something that no longer matches.
	OutActors.RemoveAll([Tag](const AActor* A) { return !A->ActorHasTag(Tag); });
}

bool ACDHudBase::IsActorOnScreen(const AActor* Actor, float Tolerance)
{
	return IsValid(Actor) && Actor->WasRecentlyRendered(FMath::Max(0.f, Tolerance));
}

void ACDHudBase::InvalidateHudCaches()
{
	Singles.Reset();
	ClassLists.Reset();
	TagLists.Reset();
}

// #500 point 5: a versioned save game with a safe write path and a written recovery rule.
//
// File format: every slot file is an envelope: magic "CDSV", envelope format, sequence, payload length, CRC32 over the header fields AND the payload,
// then the payload (= UGameplayStatics::SaveGameToMemory bytes). Each file gets a status: Missing, Valid, Corrupt (bad CRC / length / payload, or the
// header sequence differs from the save's own Sequence), UnsupportedFormat (an envelope format newer than this build), NewerVersion (valid file,
// SaveVersion newer than this build). Old plain saves (no envelope) are read as sequence 0.
//
// Write (Save):
//   refuse if the slot is read-only, the incoming object is from a newer build, or ANY copy of the slot on disk is UnsupportedFormat or NewerVersion
//   (an older build never overwrites what it cannot understand); Migrate() the incoming object; sequence = highest existing valid sequence + 1;
//   1 write <slot>_tmp and read it back
//   2 copy the current VALID main byte-for-byte to <slot>_bak and read it back. If this fails: DELETE _tmp and return false = ROLLBACK, nothing committed,
//     main untouched (a failed save is never recovered later)
//   3 write main and read it back. If this fails: return false, but the verified _tmp is KEPT as recoverable progress (main may be damaged now)
//   4 delete _tmp.
// Recovery (LoadOrCreate): among the Valid copies main, _tmp, _bak, take the highest sequence; on a tie main > _tmp > _bak. A _tmp can only be left
//   by step 3 failing (or a crash between 3 and 4), so it is newer than main only when its commit did not complete. Corrupt / older copies never
//   replace a newer valid one. An UnsupportedFormat or NewerVersion copy makes the slot read-only (per slot).
// Note: SaveGameToMemory stores every non-transient UPROPERTY of UCDSaveGame.
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/SaveGame.h"
#include "Subsystems/GameInstanceSubsystem.h"
#include "CDSaveGame.generated.h"

UCLASS(BlueprintType)
class CASTLEDEFENDER_API UCDSaveGame : public USaveGame
{
	GENERATED_BODY()

public:
	// Raise when the meaning of saved data changes, and add a step in Migrate().
	static constexpr int32 CurrentVersion = 1;

	UPROPERTY(SaveGame, BlueprintReadOnly, Category = "Save")
	int32 SaveVersion = CurrentVersion;

	// Monotonic per slot: decides which copy is newest during recovery.
	UPROPERTY(SaveGame, BlueprintReadOnly, Category = "Save")
	int64 Sequence = 0;

	UPROPERTY(SaveGame, BlueprintReadOnly, Category = "Save")
	FDateTime SavedAt;

	UPROPERTY(SaveGame) TMap<FName, int32> Ints;
	UPROPERTY(SaveGame) TMap<FName, float> Floats;
	UPROPERTY(SaveGame) TMap<FName, FString> Strings;
	UPROPERTY(SaveGame) TSet<FName> Flags;

	UFUNCTION(BlueprintCallable, Category = "Save") void SetInt(FName Key, int32 Value) { Ints.Add(Key, Value); }
	UFUNCTION(BlueprintPure, Category = "Save") int32 GetInt(FName Key, int32 Fallback = 0) const { const int32* V = Ints.Find(Key); return V ? *V : Fallback; }
	UFUNCTION(BlueprintCallable, Category = "Save") void SetFloat(FName Key, float Value) { Floats.Add(Key, Value); }
	UFUNCTION(BlueprintPure, Category = "Save") float GetFloat(FName Key, float Fallback = 0.f) const { const float* V = Floats.Find(Key); return V ? *V : Fallback; }
	UFUNCTION(BlueprintCallable, Category = "Save") void SetString(FName Key, const FString& Value) { Strings.Add(Key, Value); }
	UFUNCTION(BlueprintPure, Category = "Save") FString GetString(FName Key) const { const FString* V = Strings.Find(Key); return V ? *V : FString(); }
	UFUNCTION(BlueprintCallable, Category = "Save") void SetFlag(FName Key, bool bOn) { if (bOn) { Flags.Add(Key); } else { Flags.Remove(Key); } }
	UFUNCTION(BlueprintPure, Category = "Save") bool HasFlag(FName Key) const { return Flags.Contains(Key); }

	// Upgrades an older save to CurrentVersion (each step safe to run twice). False if the save is from a newer build (left unchanged).
	bool Migrate();
	bool IsNewerThanBuild() const { return SaveVersion > CurrentVersion; }
};

UENUM(BlueprintType)
enum class ECDSaveFileStatus : uint8
{
	Missing,
	Valid,
	Corrupt,
	UnsupportedFormat,
	NewerVersion
};

UCLASS()
class CASTLEDEFENDER_API UCDSaveSubsystem : public UGameInstanceSubsystem
{
	GENERATED_BODY()

public:
	UFUNCTION(BlueprintCallable, Category = "Save")
	UCDSaveGame* LoadOrCreate(const FString& Slot = TEXT("Main"));

	// False if anything failed or was refused; the main slot is then unchanged, or (if only the final write failed) the newest verified copy is in _tmp / _bak.
	UFUNCTION(BlueprintCallable, Category = "Save")
	bool Save(UCDSaveGame* Data, const FString& Slot = TEXT("Main"));

	UFUNCTION(BlueprintPure, Category = "Save")
	UCDSaveGame* GetCurrent() const { return Current; }

	// True when this slot holds a save from a newer game build: it is read-only, Save() refuses to write it. Other slots are unaffected.
	UFUNCTION(BlueprintPure, Category = "Save")
	bool IsReadOnly(const FString& Slot = TEXT("Main")) const { return ReadOnlySlots.Contains(Slot); }

	// One slot file, fully checked. OutData is set only for Valid and NewerVersion. Public for tests.
	static ECDSaveFileStatus Inspect(const FString& SlotFile, UCDSaveGame*& OutData);

	// Convenience: the save object if the file is Valid (or NewerVersion), else null.
	static UCDSaveGame* LoadValid(const FString& SlotFile);

	// Envelope helpers (public for tests).
	static void Wrap(const TArray<uint8>& Payload, int64 Sequence, TArray<uint8>& Out);
	static ECDSaveFileStatus Unwrap(const TArray<uint8>& File, TArray<uint8>& OutPayload, int64& OutHeaderSequence, bool& bOutLegacy);

private:
	UPROPERTY()
	TObjectPtr<UCDSaveGame> Current;

	TSet<FString> ReadOnlySlots;
};

#include "CDSaveGame.h"
#include "Kismet/GameplayStatics.h"
#include "Misc/Crc.h"
#include "Serialization/MemoryReader.h"
#include "Serialization/MemoryWriter.h"

namespace
{
	const uint32 EnvelopeMagic = 0x56534443;   // "CDSV" little-endian
	const uint32 EnvelopeFormat = 2;           // 2 = CRC over header fields + payload (format 1 was never shipped)
	const int32 HeaderFieldsSize = 4 + 4 + 8 + 4;  // magic, format, sequence, length
	const int32 HeaderSize = HeaderFieldsSize + 4;  // + CRC

	bool ReadBytes(const FString& SlotFile, TArray<uint8>& Out)
	{
		Out.Reset();
		return UGameplayStatics::DoesSaveGameExist(SlotFile, 0) && UGameplayStatics::LoadDataFromSlot(Out, SlotFile, 0) && Out.Num() > 0;
	}

	bool WriteVerified(const TArray<uint8>& Bytes, const FString& SlotFile)
	{
		TArray<uint8> Back;
		return UGameplayStatics::SaveDataToSlot(Bytes, SlotFile, 0) && ReadBytes(SlotFile, Back) && Back == Bytes;
	}

	uint32 EnvelopeCrc(const uint8* HeaderFields, const uint8* Payload, int32 PayloadLen)
	{
		return FCrc::MemCrc32(Payload, PayloadLen, FCrc::MemCrc32(HeaderFields, HeaderFieldsSize));
	}

	// The three files of one slot, in tie-break order.
	TArray<FString, TInlineAllocator<3>> SlotFiles(const FString& Slot)
	{
		return { Slot, Slot + TEXT("_tmp"), Slot + TEXT("_bak") };
	}

	bool Blocks(ECDSaveFileStatus S) { return S == ECDSaveFileStatus::UnsupportedFormat || S == ECDSaveFileStatus::NewerVersion; }
}

bool UCDSaveGame::Migrate()
{
	if (IsNewerThanBuild())
	{
		return false;
	}
	// One step per version bump, oldest first; each step must be safe to run twice. Example for a future version 2:
	// if (SaveVersion < 2) { Ints.FindOrAdd(TEXT("Gems")); SaveVersion = 2; }
	SaveVersion = CurrentVersion;
	return true;
}

void UCDSaveSubsystem::Wrap(const TArray<uint8>& Payload, int64 Sequence, TArray<uint8>& Out)
{
	Out.Reset(HeaderSize + Payload.Num());
	FMemoryWriter W(Out);
	uint32 Magic = EnvelopeMagic, Format = EnvelopeFormat, Len = Payload.Num();
	W << Magic << Format << Sequence << Len;
	uint32 Crc = EnvelopeCrc(Out.GetData(), Payload.GetData(), Payload.Num());
	W << Crc;
	Out.Append(Payload);
}

ECDSaveFileStatus UCDSaveSubsystem::Unwrap(const TArray<uint8>& File, TArray<uint8>& OutPayload, int64& OutHeaderSequence, bool& bOutLegacy)
{
	OutPayload.Reset();
	OutHeaderSequence = 0;
	bOutLegacy = false;
	if (File.Num() == 0)
	{
		return ECDSaveFileStatus::Missing;
	}
	uint32 Magic = 0;
	if (File.Num() >= 4)
	{
		FMemoryReader Peek(File);
		Peek << Magic;
	}
	if (Magic != EnvelopeMagic)
	{
		bOutLegacy = true;          // plain save from before the envelope: validity is decided by deserializing
		OutPayload = File;
		return ECDSaveFileStatus::Valid;
	}
	if (File.Num() < HeaderSize)
	{
		return ECDSaveFileStatus::Corrupt;
	}
	FMemoryReader R(File);
	uint32 Format = 0, Len = 0, Crc = 0;
	R << Magic << Format;
	if (Format > EnvelopeFormat)
	{
		return ECDSaveFileStatus::UnsupportedFormat;   // written by a newer build: protect it, never overwrite
	}
	if (Format != EnvelopeFormat)
	{
		return ECDSaveFileStatus::Corrupt;
	}
	R << OutHeaderSequence << Len << Crc;
	if ((int64)Len != (int64)File.Num() - HeaderSize || EnvelopeCrc(File.GetData(), File.GetData() + HeaderSize, Len) != Crc)
	{
		return ECDSaveFileStatus::Corrupt;
	}
	OutPayload.Append(File.GetData() + HeaderSize, Len);
	return ECDSaveFileStatus::Valid;
}

ECDSaveFileStatus UCDSaveSubsystem::Inspect(const FString& SlotFile, UCDSaveGame*& OutData)
{
	OutData = nullptr;
	TArray<uint8> File, Payload;
	if (!ReadBytes(SlotFile, File))
	{
		return ECDSaveFileStatus::Missing;
	}
	int64 HeaderSequence = 0;
	bool bLegacy = false;
	const ECDSaveFileStatus Envelope = Unwrap(File, Payload, HeaderSequence, bLegacy);
	if (Envelope != ECDSaveFileStatus::Valid)
	{
		return Envelope;
	}
	UCDSaveGame* Data = Cast<UCDSaveGame>(UGameplayStatics::LoadGameFromMemory(Payload));
	if (!Data || (!bLegacy && Data->Sequence != HeaderSequence))
	{
		return ECDSaveFileStatus::Corrupt;   // does not deserialize, or the header and the save disagree on the sequence
	}
	OutData = Data;
	return Data->IsNewerThanBuild() ? ECDSaveFileStatus::NewerVersion : ECDSaveFileStatus::Valid;
}

UCDSaveGame* UCDSaveSubsystem::LoadValid(const FString& SlotFile)
{
	UCDSaveGame* Data = nullptr;
	Inspect(SlotFile, Data);
	return Data;
}

UCDSaveGame* UCDSaveSubsystem::LoadOrCreate(const FString& Slot)
{
	ReadOnlySlots.Remove(Slot);
	UCDSaveGame* Best = nullptr;
	FString BestFile;
	bool bBlocked = false;
	for (const FString& File : SlotFiles(Slot))
	{
		UCDSaveGame* Copy = nullptr;
		const ECDSaveFileStatus Status = Inspect(File, Copy);
		if (Blocks(Status))
		{
			bBlocked = true;
		}
		if (Status == ECDSaveFileStatus::Corrupt)
		{
			UE_LOG(LogTemp, Warning, TEXT("CDSave: %s is corrupt; ignored, left on disk"), *File);
		}
		if (!Copy)
		{
			continue;
		}
		if (!Best || Copy->Sequence > Best->Sequence)   // strict >: on a tie the earlier file (main > _tmp > _bak) wins
		{
			Best = Copy;
			BestFile = File;
		}
	}
	if (bBlocked)
	{
		ReadOnlySlots.Add(Slot);
		UE_LOG(LogTemp, Error, TEXT("CDSave: slot %s holds a save this build cannot write safely (newer format or version): read-only"), *Slot);
	}
	else if (Best)
	{
		if (BestFile != Slot)
		{
			UE_LOG(LogTemp, Warning, TEXT("CDSave: recovered slot %s from %s (sequence %lld)"), *Slot, *BestFile, Best->Sequence);
		}
		Best->Migrate();
	}
	if (!Best)
	{
		Best = Cast<UCDSaveGame>(UGameplayStatics::CreateSaveGameObject(UCDSaveGame::StaticClass()));
	}
	Current = Best;
	return Best;
}

bool UCDSaveSubsystem::Save(UCDSaveGame* Data, const FString& Slot)
{
	if (!Data)
	{
		return false;
	}
	if (ReadOnlySlots.Contains(Slot) || Data->IsNewerThanBuild())
	{
		UE_LOG(LogTemp, Error, TEXT("CDSave: refusing to write %s: read-only or data from a newer build"), *Slot);
		return false;
	}
	// Look at what is on disk now, whatever this session loaded: never overwrite what this build cannot understand; continue the sequence.
	int64 MaxSequence = 0;
	for (const FString& File : SlotFiles(Slot))
	{
		UCDSaveGame* Existing = nullptr;
		const ECDSaveFileStatus Status = Inspect(File, Existing);
		if (Blocks(Status))
		{
			ReadOnlySlots.Add(Slot);
			UE_LOG(LogTemp, Error, TEXT("CDSave: %s is from a newer build (format or version); refusing to overwrite slot %s"), *File, *Slot);
			return false;
		}
		if (Existing)
		{
			MaxSequence = FMath::Max(MaxSequence, Existing->Sequence);
		}
	}
	if (!Data->Migrate())
	{
		return false;
	}
	Data->Sequence = MaxSequence + 1;
	Data->SavedAt = FDateTime::UtcNow();

	// 0. serialize, check it reads back as a save, wrap with header + checksum
	TArray<uint8> Payload, File;
	if (!UGameplayStatics::SaveGameToMemory(Data, Payload) || !Cast<UCDSaveGame>(UGameplayStatics::LoadGameFromMemory(Payload)))
	{
		UE_LOG(LogTemp, Error, TEXT("CDSave: %s did not serialize cleanly; nothing written"), *Slot);
		return false;
	}
	Wrap(Payload, Data->Sequence, File);
	// 1. temp copy, read back
	const FString Tmp = Slot + TEXT("_tmp");
	if (!WriteVerified(File, Tmp))
	{
		UGameplayStatics::DeleteGameInSlot(Tmp, 0);
		UE_LOG(LogTemp, Error, TEXT("CDSave: could not write %s; nothing committed"), *Tmp);
		return false;
	}
	// 2. back up the current main byte-for-byte, only if it is Valid. Failure = rollback: _tmp deleted, nothing committed.
	TArray<uint8> Old;
	UCDSaveGame* OldData = nullptr;
	if (Inspect(Slot, OldData) == ECDSaveFileStatus::Valid && ReadBytes(Slot, Old))
	{
		if (!WriteVerified(Old, Slot + TEXT("_bak")))
		{
			UGameplayStatics::DeleteGameInSlot(Tmp, 0);
			UE_LOG(LogTemp, Error, TEXT("CDSave: backup of %s failed; rolled back, main untouched, nothing committed"), *Slot);
			return false;
		}
	}
	else if (UGameplayStatics::DoesSaveGameExist(Slot, 0))
	{
		UE_LOG(LogTemp, Warning, TEXT("CDSave: %s was corrupt; keeping the existing %s_bak, overwriting main"), *Slot, *Slot);
	}
	// 3. main. Failure: the verified _tmp is kept as recoverable progress (documented in the header).
	if (!WriteVerified(File, Slot))
	{
		UE_LOG(LogTemp, Error, TEXT("CDSave: writing %s failed; the verified %s (sequence %lld) is kept and will be recovered"), *Slot, *Tmp, Data->Sequence);
		return false;
	}
	// 4. commit done
	UGameplayStatics::DeleteGameInSlot(Tmp, 0);
	Current = Data;
	return true;
}

// #500 point 8: reuse short-lived actors (bolts, blood bursts, gore pieces) instead of spawning and destroying one per hit.
// Blueprint: "Acquire Pooled Actor" instead of SpawnActor, "Release Pooled Actor" instead of DestroyActor / SetLifeSpan.
//
// RESET CONTRACT: only classes that implement ICDPoolable are pooled. OnAcquired must bring the actor back to a just-spawned state
// (BeginPlay does NOT run again): health / damage state, velocity, projectile movement, ignored actors, attached children, effects, delegates.
// OnReleased must stop everything it started, INCLUDING timers set by its components or by other objects for it: the pool only clears timers bound
// to the actor object itself (ClearAllTimersForObject). The pool also: clears the life span, zeroes physics velocity, stops and DEACTIVATES active
// movement components, turns off every component tick (actor tick and component ticks are separate in Unreal), and remembers the actor's own
// hidden / tick / collision values before turning collision off, tick off and hiding it. On Acquire it re-enables exactly those component ticks and
// re-activates exactly those movers, calls OnAcquired while the actor is still hidden and without collision, then puts back the actor's own
// visibility and tick, and its collision LAST (so overlaps never see old state).
// Component Activate / Deactivate can run callbacks that call back into the pool: no record reference is held across them; the record is looked
// up again and its generation checked after every such call.
// A class without ICDPoolable is spawned normally by Acquire and destroyed by Release: never pooled.
//
// Lifecycle per actor: Active -> (Release with delay) PendingRelease -> Pooled -> (Acquire) Active ... Each Acquire bumps a generation number;
// a delayed release only parks the actor if it is still the same generation and still pending, so an old timer can never park a reused actor.
//
// Niagara / Cascade effects: use the engine's pooling instead (PoolingMethod = AutoRelease on SpawnSystemAtLocation / SpawnEmitterAtLocation).
#pragma once

#include "CoreMinimal.h"
#include "Engine/TimerHandle.h"
#include "Subsystems/WorldSubsystem.h"
#include "UObject/Interface.h"
#include "CDPoolSubsystem.generated.h"

UINTERFACE(BlueprintType, Blueprintable)
class CASTLEDEFENDER_API UCDPoolable : public UInterface
{
	GENERATED_BODY()
};

class CASTLEDEFENDER_API ICDPoolable
{
	GENERATED_BODY()

public:
	// Reset to a just-spawned state (see the contract above).
	UFUNCTION(BlueprintNativeEvent, Category = "Pool")
	void OnAcquired();

	// Stop everything this actor started (sounds, effects, timers it owns elsewhere).
	UFUNCTION(BlueprintNativeEvent, Category = "Pool")
	void OnReleased();
};

UCLASS()
class CASTLEDEFENDER_API UCDPoolSubsystem : public UWorldSubsystem
{
	GENERATED_BODY()

public:
	UFUNCTION(BlueprintCallable, Category = "Pool", meta = (DeterminesOutputType = "Class"))
	AActor* AcquirePooledActor(TSubclassOf<AActor> Class, const FTransform& Transform);

	// Active actor only (a second release, or a release of something this pool did not hand out, is ignored with a warning).
	// LifeSeconds > 0: release after that delay. Non-poolable classes are destroyed.
	UFUNCTION(BlueprintCallable, Category = "Pool")
	void ReleasePooledActor(AActor* Actor, float LifeSeconds = 0.f);

	UFUNCTION(BlueprintPure, Category = "Pool")
	int32 GetPooledCount(TSubclassOf<AActor> Class) const;

	UFUNCTION(BlueprintPure, Category = "Pool")
	static bool IsPoolable(TSubclassOf<AActor> Class);

	// Per class, at most this many are kept parked; extra releases are destroyed.
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Pool", meta = (ClampMin = "0"))
	int32 MaxPerClass = 64;

	virtual void Deinitialize() override;

protected:
	virtual bool DoesSupportWorldType(const EWorldType::Type WorldType) const override;

private:
	enum class EState : uint8 { Active, PendingRelease, Pooled };
	struct FRecord
	{
		EState State = EState::Active;
		uint32 Generation = 0;
		FTimerHandle Timer;
		TArray<TWeakObjectPtr<UActorComponent>> TicksOff;      // component ticks the pool turned off
		TArray<TWeakObjectPtr<UActorComponent>> MoversOff;     // movement components the pool deactivated
		bool bWasHidden = false, bHadCollision = true, bHadTick = true;   // the actor's own values before parking
	};
	// Built in a local while engine calls run (they may call back into the pool), then moved into the record after a fresh lookup.
	struct FParkedState
	{
		TArray<TWeakObjectPtr<UActorComponent>> TicksOff, MoversOff;
		bool bWasHidden = false, bHadCollision = true, bHadTick = true;
	};

	void Park(AActor* Actor, uint32 ExpectedGeneration);
	void PurgeInvalid(UClass* Class);
	FRecord* FindCurrent(AActor* Actor, uint32 Generation, EState State);
	static void ResetEngineState(AActor* Actor, FParkedState& Out);
	static void RestoreComponents(const FParkedState& State);

	TMap<TWeakObjectPtr<AActor>, FRecord> Records;
	TMap<TWeakObjectPtr<UClass>, TArray<TWeakObjectPtr<AActor>>> Free;
	uint32 NextGeneration = 1;
};

#include "CDPoolSubsystem.h"
#include "CDStats.h"
#include "Components/PrimitiveComponent.h"
#include "Engine/World.h"
#include "GameFramework/MovementComponent.h"
#include "TimerManager.h"

DECLARE_CYCLE_STAT(TEXT("Pool Acquire"), STAT_CD_PoolAcquire, STATGROUP_CastleDefender);
DECLARE_DWORD_COUNTER_STAT(TEXT("Pool parked actors"), STAT_CD_PoolParked, STATGROUP_CastleDefender);

void ICDPoolable::OnAcquired_Implementation() {}
void ICDPoolable::OnReleased_Implementation() {}

bool UCDPoolSubsystem::DoesSupportWorldType(const EWorldType::Type WorldType) const
{
	return WorldType == EWorldType::Game || WorldType == EWorldType::PIE;
}

bool UCDPoolSubsystem::IsPoolable(TSubclassOf<AActor> Class)
{
	return *Class && Class->ImplementsInterface(UCDPoolable::StaticClass());
}

void UCDPoolSubsystem::Deinitialize()
{
	if (UWorld* World = GetWorld())
	{
		for (TPair<TWeakObjectPtr<AActor>, FRecord>& Pair : Records)
		{
			World->GetTimerManager().ClearTimer(Pair.Value.Timer);
		}
	}
	Records.Reset();
	Free.Reset();
	Super::Deinitialize();
}

UCDPoolSubsystem::FRecord* UCDPoolSubsystem::FindCurrent(AActor* Actor, uint32 Generation, EState State)
{
	FRecord* R = Records.Find(Actor);
	return (R && R->Generation == Generation && R->State == State) ? R : nullptr;
}

void UCDPoolSubsystem::ResetEngineState(AActor* Actor, FParkedState& Out)
{
	// Works on a local struct only: Deactivate / SetComponentTickEnabled may run callbacks that change the pool's maps.
	if (UWorld* World = Actor->GetWorld())
	{
		World->GetTimerManager().ClearAllTimersForObject(Actor);
	}
	Actor->SetLifeSpan(0.f);
	Out.bWasHidden = Actor->IsHidden();
	Out.bHadCollision = Actor->GetActorEnableCollision();
	Out.bHadTick = Actor->IsActorTickEnabled();
	TInlineComponentArray<UActorComponent*> Components(Actor);
	for (UActorComponent* Comp : Components)
	{
		if (!IsValid(Comp))
		{
			continue;
		}
		if (UMovementComponent* Mover = Cast<UMovementComponent>(Comp))
		{
			Mover->StopMovementImmediately();
			if (Mover->IsActive())
			{
				Out.MoversOff.Add(Mover);
				Mover->Deactivate();
			}
		}
		if (Comp->IsComponentTickEnabled())
		{
			Out.TicksOff.Add(Comp);
			Comp->SetComponentTickEnabled(false);
		}
	}
	if (UPrimitiveComponent* Root = Cast<UPrimitiveComponent>(Actor->GetRootComponent()))
	{
		if (Root->IsSimulatingPhysics())
		{
			Root->SetPhysicsLinearVelocity(FVector::ZeroVector);
			Root->SetPhysicsAngularVelocityInDegrees(FVector::ZeroVector);
		}
	}
}

void UCDPoolSubsystem::RestoreComponents(const FParkedState& State)
{
	for (const TWeakObjectPtr<UActorComponent>& W : State.TicksOff)
	{
		if (UActorComponent* Comp = W.Get()) { Comp->SetComponentTickEnabled(true); }
	}
	for (const TWeakObjectPtr<UActorComponent>& W : State.MoversOff)
	{
		if (UActorComponent* Comp = W.Get()) { Comp->Activate(true); }
	}
}

void UCDPoolSubsystem::PurgeInvalid(UClass* Class)
{
	if (TArray<TWeakObjectPtr<AActor>>* List = Free.Find(Class))
	{
		List->RemoveAll([](const TWeakObjectPtr<AActor>& W) { return !W.IsValid(); });
	}
	for (auto It = Records.CreateIterator(); It; ++It)
	{
		if (!It.Key().IsValid())
		{
			It.RemoveCurrent();
		}
	}
}

AActor* UCDPoolSubsystem::AcquirePooledActor(TSubclassOf<AActor> Class, const FTransform& Transform)
{
	CD_SCOPE(STAT_CD_PoolAcquire);
	UWorld* World = GetWorld();
	if (!World || !*Class)
	{
		return nullptr;
	}
	if (!IsPoolable(Class))
	{
		return World->SpawnActor<AActor>(Class, Transform);   // not poolable: plain spawn, Release will destroy it
	}

	PurgeInvalid(*Class);
	AActor* Actor = nullptr;
	if (TArray<TWeakObjectPtr<AActor>>* List = Free.Find(*Class))
	{
		while (List->Num() > 0 && !Actor)
		{
			AActor* Candidate = List->Pop().Get();
			const FRecord* Rec = Candidate ? Records.Find(Candidate) : nullptr;
			if (Rec && Rec->State == EState::Pooled && Candidate->GetWorld() == World)
			{
				Actor = Candidate;
			}
		}
	}
	const bool bFromPool = Actor != nullptr;
	if (bFromPool)
	{
		Actor->SetActorTransform(Transform, false, nullptr, ETeleportType::ResetPhysics);   // still hidden, still without collision
	}
	else
	{
		FActorSpawnParameters Params;
		Params.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AlwaysSpawn;
		Actor = World->SpawnActor<AActor>(Class, Transform, Params);
		if (!Actor)
		{
			return nullptr;
		}
	}

	// Mark it ours BEFORE any callback, and move the parked state out of the record into a local.
	FParkedState Parked;
	uint32 Generation = 0;
	{
		FRecord& Rec = Records.FindOrAdd(Actor);   // nothing between here and the end of this block can call out
		World->GetTimerManager().ClearTimer(Rec.Timer);
		Parked.TicksOff = MoveTemp(Rec.TicksOff);
		Parked.MoversOff = MoveTemp(Rec.MoversOff);
		Parked.bWasHidden = Rec.bWasHidden;
		Parked.bHadCollision = Rec.bHadCollision;
		Parked.bHadTick = Rec.bHadTick;
		Rec.State = EState::Active;
		Rec.Generation = Generation = NextGeneration++;
	}
	if (!bFromPool)
	{
		ICDPoolable::Execute_OnAcquired(Actor);   // fresh spawn: its own BeginPlay state, nothing to restore
		return Actor;
	}
	RestoreComponents(Parked);                 // may run callbacks
	ICDPoolable::Execute_OnAcquired(Actor);    // reset gameplay state while still hidden and without collision
	if (!IsValid(Actor) || !FindCurrent(Actor, Generation, EState::Active))
	{
		return IsValid(Actor) ? Actor : nullptr;   // a callback released or destroyed it: leave it as that left it
	}
	Actor->SetActorHiddenInGame(Parked.bWasHidden);
	Actor->SetActorTickEnabled(Parked.bHadTick);
	Actor->SetActorEnableCollision(Parked.bHadCollision);   // last
	return Actor;
}

void UCDPoolSubsystem::ReleasePooledActor(AActor* Actor, float LifeSeconds)
{
	UWorld* World = GetWorld();
	if (!IsValid(Actor) || !World)
	{
		return;
	}
	if (Actor->GetWorld() != World)
	{
		UE_LOG(LogTemp, Warning, TEXT("CDPool: %s belongs to another world; not released here"), *Actor->GetName());
		return;
	}
	if (!IsPoolable(Actor->GetClass()))
	{
		if (LifeSeconds > 0.f)
		{
			Actor->SetLifeSpan(LifeSeconds);
		}
		else
		{
			Actor->Destroy();
		}
		return;
	}
	FRecord* Rec = Records.Find(Actor);
	if (!Rec || Rec->State != EState::Active)
	{
		UE_LOG(LogTemp, Warning, TEXT("CDPool: %s released twice or not handed out by the pool; ignored"), *Actor->GetName());
		return;
	}
	const uint32 Generation = Rec->Generation;
	Rec->State = EState::PendingRelease;
	if (LifeSeconds > 0.f)
	{
		TWeakObjectPtr<AActor> Weak(Actor);
		World->GetTimerManager().SetTimer(Rec->Timer, FTimerDelegate::CreateWeakLambda(this, [this, Weak, Generation]()
		{
			if (AActor* A = Weak.Get())
			{
				Park(A, Generation);
			}
		}), LifeSeconds, false);
		return;
	}
	Park(Actor, Generation);
}

void UCDPoolSubsystem::Park(AActor* Actor, uint32 ExpectedGeneration)
{
	// Never hold an FRecord pointer across anything that can call out (OnReleased, Deactivate, tick changes, Destroy): look it up again.
	if (FRecord* Rec = FindCurrent(Actor, ExpectedGeneration, EState::PendingRelease))
	{
		if (UWorld* World = GetWorld()) { World->GetTimerManager().ClearTimer(Rec->Timer); }
	}
	else
	{
		return;   // reacquired since the release was scheduled, or already parked
	}
	ICDPoolable::Execute_OnReleased(Actor);
	if (!IsValid(Actor))
	{
		Records.Remove(Actor);   // OnReleased destroyed it
		return;
	}
	PurgeInvalid(Actor->GetClass());
	if (!FindCurrent(Actor, ExpectedGeneration, EState::PendingRelease))
	{
		return;   // OnReleased re-acquired it
	}
	if (GetPooledCount(Actor->GetClass()) >= MaxPerClass)
	{
		Records.Remove(Actor);
		Actor->Destroy();
		return;
	}
	FParkedState Parked;
	ResetEngineState(Actor, Parked);            // may run callbacks: no record reference held
	if (!IsValid(Actor))
	{
		Records.Remove(Actor);
		return;
	}
	if (!FindCurrent(Actor, ExpectedGeneration, EState::PendingRelease))
	{
		RestoreComponents(Parked);              // a callback re-acquired it meanwhile: undo our component changes and stop
		return;
	}
	Actor->SetActorEnableCollision(false);      // collision off first, visibility last
	Actor->SetActorTickEnabled(false);
	Actor->SetActorHiddenInGame(true);
	FRecord* Rec = FindCurrent(Actor, ExpectedGeneration, EState::PendingRelease);
	if (!Rec)
	{
		return;
	}
	Rec->TicksOff = MoveTemp(Parked.TicksOff);
	Rec->MoversOff = MoveTemp(Parked.MoversOff);
	Rec->bWasHidden = Parked.bWasHidden;
	Rec->bHadCollision = Parked.bHadCollision;
	Rec->bHadTick = Parked.bHadTick;
	Rec->State = EState::Pooled;
	Free.FindOrAdd(Actor->GetClass()).Add(Actor);
	int32 ParkedCount = 0;
	for (const TPair<TWeakObjectPtr<UClass>, TArray<TWeakObjectPtr<AActor>>>& Pair : Free) { ParkedCount += Pair.Value.Num(); }
	SET_DWORD_STAT(STAT_CD_PoolParked, ParkedCount);
}

int32 UCDPoolSubsystem::GetPooledCount(TSubclassOf<AActor> Class) const
{
	const TArray<TWeakObjectPtr<AActor>>* List = Free.Find(*Class);
	if (!List)
	{
		return 0;
	}
	int32 Count = 0;
	for (const TWeakObjectPtr<AActor>& W : *List)
	{
		Count += W.IsValid() ? 1 : 0;
	}
	return Count;
}

IMPLEMENT_SIMPLE_AUTOMATION_TEST(FCDSaveTest, "CastleDefender.Save.SafeWritePath",
	EAutomationTestFlags::EditorContext | EAutomationTestFlags::EngineFilter)

bool FCDSaveTest::RunTest(const FString& Parameters)
{
	// A fresh slot name per run: the test can never delete real or older test data.
	const FString Slot = FString::Printf(TEXT("CDTest_%s"), *FGuid::NewGuid().ToString(EGuidFormats::Digits));
	const FString Tmp = Slot + TEXT("_tmp"), Bak = Slot + TEXT("_bak");
	auto Clean = [&]() { for (const FString& F : { Slot, Tmp, Bak }) { UGameplayStatics::DeleteGameInSlot(F, 0); } };
	auto Make = [](int32 Gold, int32 Version = UCDSaveGame::CurrentVersion)
	{
		UCDSaveGame* D = Cast<UCDSaveGame>(UGameplayStatics::CreateSaveGameObject(UCDSaveGame::StaticClass()));
		D->SetInt(TEXT("Gold"), Gold);
		D->SaveVersion = Version;
		return D;
	};
	// Writes a valid enveloped file directly (to set up recovery cases).
	auto WriteRaw = [](UCDSaveGame* D, int64 Seq, const FString& File)
	{
		D->Sequence = Seq;
		TArray<uint8> Payload, Out;
		UGameplayStatics::SaveGameToMemory(D, Payload);
		UCDSaveSubsystem::Wrap(Payload, Seq, Out);
		return UGameplayStatics::SaveDataToSlot(Out, File, 0);
	};
	// Null-safe check of one file's Gold.
	auto ExpectGold = [this](const FString& What, const FString& File, int32 Gold)
	{
		const UCDSaveGame* L = UCDSaveSubsystem::LoadValid(File);
		if (TestNotNull(*(What + TEXT(" (loads)")), L)) { TestEqual(*What, L->GetInt(TEXT("Gold"), -1), Gold); }
	};
	auto ExpectLoaded = [this](const FString& What, const UCDSaveGame* L, int32 Gold)
	{
		if (TestNotNull(*(What + TEXT(" (object)")), L)) { TestEqual(*What, L->GetInt(TEXT("Gold"), -1), Gold); }
	};
	const TArray<uint8> Garbage = { 1, 2, 3, 4, 5, 6, 7, 8 };
	UCDSaveSubsystem* Saves = NewObject<UCDSaveSubsystem>(GetTransientPackage());

	Clean();
	ExpectLoaded(TEXT("missing save -> fresh default"), Saves->LoadOrCreate(Slot), -1);

	TestTrue(TEXT("first save works"), Saves->Save(Make(10), Slot));
	TestTrue(TEXT("second save works"), Saves->Save(Make(20), Slot));
	ExpectGold(TEXT("main holds the newest"), Slot, 20);
	ExpectGold(TEXT("backup holds the previous"), Bak, 10);
	TestFalse(TEXT("temp file removed after commit"), UGameplayStatics::DoesSaveGameExist(Tmp, 0));

	// checksum: one flipped byte makes the file invalid
	TArray<uint8> Bytes;
	UGameplayStatics::LoadDataFromSlot(Bytes, Slot, 0);
	if (TestTrue(TEXT("main file has bytes"), Bytes.Num() > 30))
	{
		Bytes.Last() ^= 0xFF;
		UGameplayStatics::SaveDataToSlot(Bytes, Slot, 0);
		TestNull(TEXT("flipped byte -> invalid (CRC)"), UCDSaveSubsystem::LoadValid(Slot));
	}
	ExpectLoaded(TEXT("corrupt main -> verified backup is loaded"), Saves->LoadOrCreate(Slot), 10);
	TestTrue(TEXT("save over a corrupt main works"), Saves->Save(Make(30), Slot));
	ExpectGold(TEXT("corrupt main did not replace the good backup"), Bak, 10);

	UGameplayStatics::SaveDataToSlot(Garbage, Bak, 0);
	ExpectLoaded(TEXT("corrupt backup never replaces a valid main"), Saves->LoadOrCreate(Slot), 30);

	// Bug B: an interrupted save leaves a verified _tmp that is newer than main
	Clean();
	WriteRaw(Make(40), 4, Slot);
	WriteRaw(Make(50), 5, Tmp);
	ExpectLoaded(TEXT("_tmp newer than main (commit interrupted) -> _tmp recovered"), Saves->LoadOrCreate(Slot), 50);
	Clean();
	WriteRaw(Make(60), 6, Slot);
	WriteRaw(Make(55), 5, Tmp);
	ExpectLoaded(TEXT("stale _tmp older than main -> main wins"), Saves->LoadOrCreate(Slot), 60);
	Clean();
	WriteRaw(Make(70), 7, Tmp);
	ExpectLoaded(TEXT("first save interrupted, only _tmp exists -> _tmp recovered"), Saves->LoadOrCreate(Slot), 70);
	TestTrue(TEXT("saving after recovery continues the sequence"), Saves->Save(Make(71), Slot));
	const UCDSaveGame* AfterRecovery = UCDSaveSubsystem::LoadValid(Slot);
	if (TestNotNull(TEXT("main after recovery save"), AfterRecovery)) { TestTrue(TEXT("sequence > 7"), AfterRecovery->Sequence > 7); }

	// Bug A: a fresh object must not overwrite a newer build's save on disk, even without LoadOrCreate first
	Clean();
	WriteRaw(Make(99, UCDSaveGame::CurrentVersion + 1), 3, Slot);
	UCDSaveSubsystem* Fresh = NewObject<UCDSaveSubsystem>(GetTransientPackage());
	TestFalse(TEXT("fresh object over a newer on-disk save is refused"), Fresh->Save(Make(1), Slot));
	const UCDSaveGame* StillNewer = UCDSaveSubsystem::LoadValid(Slot);
	if (TestNotNull(TEXT("newer save still there"), StillNewer)) { TestEqual(TEXT("newer version kept"), StillNewer->SaveVersion, UCDSaveGame::CurrentVersion + 1); }
	TestTrue(TEXT("that slot is now read-only"), Fresh->IsReadOnly(Slot));
	const FString Other = Slot + TEXT("B");
	TestTrue(TEXT("another slot is still writable (read-only is per slot)"), Fresh->Save(Make(2), Other) && !Fresh->IsReadOnly(Other));
	for (const FString& F : { Other, Other + TEXT("_tmp"), Other + TEXT("_bak") }) { UGameplayStatics::DeleteGameInSlot(F, 0); }

	// Rev 4: an envelope format newer than this build is protected, not treated as invalid
	Clean();
	{
		TArray<uint8> Payload, Env;
		UCDSaveGame* Future = Make(77);
		Future->Sequence = 2;
		UGameplayStatics::SaveGameToMemory(Future, Payload);
		UCDSaveSubsystem::Wrap(Payload, 2, Env);
		Env[4] = 0x7F;   // format field (bytes 4..7) -> a future format
		UGameplayStatics::SaveDataToSlot(Env, Slot, 0);
		UCDSaveGame* Seen = nullptr;
		TestEqual(TEXT("future envelope format -> UnsupportedFormat"), (int32)UCDSaveSubsystem::Inspect(Slot, Seen), (int32)ECDSaveFileStatus::UnsupportedFormat);
		UCDSaveSubsystem* Older = NewObject<UCDSaveSubsystem>(GetTransientPackage());
		Older->LoadOrCreate(Slot);
		TestTrue(TEXT("future format makes the slot read-only"), Older->IsReadOnly(Slot));
		TestFalse(TEXT("future format is never overwritten"), Older->Save(Make(1), Slot));
		TArray<uint8> After;
		UGameplayStatics::LoadDataFromSlot(After, Slot, 0);
		TestTrue(TEXT("future file bytes unchanged"), After == Env);
	}

	// Rev 4: header sequence must match the save's own sequence; a changed header byte fails the checksum
	Clean();
	{
		TArray<uint8> Payload, Env;
		UCDSaveGame* D = Make(88);
		D->Sequence = 3;
		UGameplayStatics::SaveGameToMemory(D, Payload);
		UCDSaveSubsystem::Wrap(Payload, 9, Env);   // header says 9, payload says 3
		UGameplayStatics::SaveDataToSlot(Env, Slot, 0);
		UCDSaveGame* Seen = nullptr;
		TestEqual(TEXT("header / payload sequence mismatch -> Corrupt"), (int32)UCDSaveSubsystem::Inspect(Slot, Seen), (int32)ECDSaveFileStatus::Corrupt);
		UCDSaveSubsystem::Wrap(Payload, 3, Env);
		Env[8] ^= 0x01;                             // flip a bit inside the header sequence field
		UGameplayStatics::SaveDataToSlot(Env, Slot, 0);
		TestEqual(TEXT("changed header byte -> Corrupt (CRC covers the header)"), (int32)UCDSaveSubsystem::Inspect(Slot, Seen), (int32)ECDSaveFileStatus::Corrupt);
	}
	// Not automatable here: "backup write fails after _tmp succeeds" (needs a failing file system). By design that path deletes _tmp and
	// returns false, so the failed save is never recovered later; check it once by making <slot>_bak read-only on disk and saving.

	// Bug F: an old-version object passed straight to Save is migrated, not just relabelled
	Clean();
	UCDSaveGame* Old = Make(5, 0);
	TestTrue(TEXT("old-version object saves"), Saves->Save(Old, Slot));
	TestEqual(TEXT("it was migrated to the current version"), Old->SaveVersion, UCDSaveGame::CurrentVersion);

	UCDSaveGame* M = Make(5, 1);
	TestTrue(TEXT("migration runs"), M->Migrate());
	TestTrue(TEXT("migration is repeatable"), M->Migrate() && M->SaveVersion == UCDSaveGame::CurrentVersion && M->GetInt(TEXT("Gold")) == 5);

	Clean();
	return true;
}

TArray<FString> UCDEditorTools::FindBlueprintsWithErrors(const FString& Path, bool bRecompile)
{
	TArray<FString> Lines;
	for (UBlueprint* BP : LoadBlueprints(Path))
	{
		if (bRecompile)
		{
			FKismetEditorUtilities::CompileBlueprint(BP, EBlueprintCompileOptions::SkipSave);
		}
		int32 NodeErrors = 0;
		for (const UEdGraph* Graph : AllGraphs(BP))
		{
			for (const UEdGraphNode* Node : Graph->Nodes)
			{
				if (Node && Node->bHasCompilerMessage && Node->ErrorType <= EMessageSeverity::Error)
				{
					++NodeErrors;
				}
			}
		}
		if (BP->Status == BS_Error || NodeErrors > 0)
		{
			Lines.Add(FString::Printf(TEXT("%s | status %d | nodes with errors %d"), *BP->GetPathName(), (int32)BP->Status, NodeErrors));
		}
	}
	for (const FString& L : Lines) { UE_LOG(LogTemp, Warning, TEXT("CD errors: %s"), *L); }
	return Lines;
}

TArray<FString> UCDEditorTools::FindExpensiveCalls(const FString& Path)
{
	static const TSet<FName> Expensive = {
		TEXT("GetAllActorsOfClass"), TEXT("GetAllActorsWithTag"), TEXT("GetAllActorsOfClassWithTag"), TEXT("GetActorOfClass"),
		TEXT("GetAllActorsWithInterface"), TEXT("GetAllWidgetsOfClass"), TEXT("LineOfSightTo"), TEXT("GetComponentsByClass"),
		TEXT("LineTraceSingle"), TEXT("LineTraceMulti"), TEXT("SphereOverlapActors"), TEXT("FindNearestActor") };
	TArray<FString> Lines;
	for (UBlueprint* BP : LoadBlueprints(Path))
	{
		for (UEdGraph* Graph : AllGraphs(BP))
		{
			for (UEdGraphNode* Node : Graph->Nodes)
			{
				const UK2Node_CallFunction* Call = Cast<UK2Node_CallFunction>(Node);
				if (Call && Expensive.Contains(Call->FunctionReference.GetMemberName()))
				{
					Lines.Add(FString::Printf(TEXT("%s | %s | %s | %s"), *BP->GetName(), *Graph->GetName(), *Node->GetName(), *Call->FunctionReference.GetMemberName().ToString()));
				}
			}
		}
	}
	for (const FString& L : Lines) { UE_LOG(LogTemp, Display, TEXT("CD expensive: %s"), *L); }
	return Lines;
}

BEL, PL, GE, EA = unreal.BlueprintEditorLibrary, unreal.BlueprintGraphPinLibrary, unreal.BlueprintGraphEditor, unreal.EditorAssetLibrary
P = '/Game/CastleDefender/Gameplay/Castle/'
HUDS = ['BP_ZombieHUD', 'BP_MinimapHUD', 'BP_SkillTreeHUD', 'BP_BarracksHUD', 'BP_ArmyHUD', 'BP_ControlsHUD', 'BP_QueueHUD', 'BP_HeroHUD',
        'BP_CommanderHUD', 'BP_ArmoryHUD', 'BP_NoticeHUD']
MAP = {  # engine title -> (cached function, input pins, output pins)
    'GetActorOfClass': ('GetCachedActorOfClass', ['ActorClass'], ['ReturnValue']),
    'GetAllActorsOfClass': ('GetCachedActorsOfClass', ['ActorClass'], ['OutActors']),
    'GetAllActorsWithTag': ('GetCachedActorsWithTag', ['Tag'], ['OutActors']),
}
BASE = unreal.load_class(None, '/Script/CastleDefender.CDHudBase')
if not BASE: print('STOP: /Script/CastleDefender.CDHudBase not loaded. Build the C++ module first. Nothing changed.'); raise SystemExit

def inherits(bp):
    return unreal.MathLibrary.class_is_child_of(bp.generated_class(), BASE)

def pin(n, name, out):
    p = BEL.find_output_pin(n, name) if out else BEL.find_input_pin(n, name)
    return p if p and PL.is_valid(p) else None

def title(n): return str(n.get_node_title()).replace(' ', '').replace('\n', '')

def errors(bp):
    BEL.compile_blueprint(bp)
    return sorted(x.get_name() for x in GE.get_graph_editor(BEL.find_event_graph(bp)).list_nodes_with_errors())

total = 0
for h in HUDS:
    if ONLY and h != ONLY: continue
    bp = unreal.load_asset(P + h)
    if not bp: print(h, 'missing'); continue
    ge = GE.get_graph_editor(BEL.find_event_graph(bp))
    targets = [n for n in ge.list_all_nodes() if n.get_class().get_name() == 'K2Node_CallFunction' and title(n) in MAP and n.get_name() not in SKIP]
    print('%-16s %d to swap: %s' % (h, len(targets), ', '.join('%s(%s)' % (n.get_name(), title(n)) for n in targets)))
    total += len(targets)
    if not APPLY or not targets: continue
    if not inherits(bp):
        print('   STOP %s: does not inherit CDHudBase yet (run reparent_hud_ue.py on BP_ZombieHUD first). Nothing changed here.' % h); continue
    before = errors(bp)
    src = unreal.SystemLibrary.convert_to_absolute_path(unreal.Paths.project_content_dir()) + (P + h)[len('/Game/'):] + '.uasset'
    dst = unreal.SystemLibrary.convert_to_absolute_path(unreal.Paths.project_saved_dir()) + 'bak500/' + h + '.uasset.hud2'
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    if not os.path.exists(dst): shutil.copy2(src, dst)
    fails = []
    for old in targets:
        fn, ins, outs = MAP[title(old)]
        new = ge.add_call_function_node('/Script/CastleDefender.CDHudBase:' + fn)
        if not new: fails.append(old.get_name() + ' (could not create ' + fn + ')'); continue
        try: BEL.set_node_pos(new, unreal.IntPoint(old.get_editor_property('node_pos_x'), old.get_editor_property('node_pos_y') + 40))
        except Exception: pass
        for name in ins:                                   # value or link of the class / tag pin first: it types the outputs
            op, np = pin(old, name, False), pin(new, name, False)
            if not op or not np: fails.append('%s.%s (no pin)' % (old.get_name(), name)); continue
            src_links = PL.list_connected_pins(op)
            if src_links:
                for s in src_links:
                    if not PL.try_create_connection(s, np): fails.append('%s.%s (in link)' % (old.get_name(), name))
            elif not PL.set_pin_value(np, PL.get_pin_value(op)):
                fails.append('%s.%s = %s (value refused)' % (old.get_name(), name, PL.get_pin_value(op)))
        ex_old, ex_new = pin(old, 'execute', False), pin(new, 'execute', False)
        if ex_old and ex_new:
            for s in PL.list_connected_pins(ex_old):
                if not PL.try_create_connection(s, ex_new): fails.append(old.get_name() + '.execute (link)')
        elif ex_old and PL.list_connected_pins(ex_old):
            fails.append(old.get_name() + '.execute (new node has no exec pin)')
        for name in ['then'] + outs:
            op, np = pin(old, name, True), pin(new, name, True)
            if not op: continue
            if not np: fails.append('%s.%s (no out pin)' % (old.get_name(), name)); continue
            for t in PL.list_connected_pins(op):
                PL.break_single_pin_link(op, t)
                if not PL.try_create_connection(np, t): fails.append('%s.%s -> %s (out link)' % (old.get_name(), name, PL.get_owning_node(t).get_name()))
        ge.remove_nodes([old])
    after = errors(bp); new_err = [e for e in after if e not in before]
    print('   RESULT %s | link/value failures %s | new errors %s | old errors %d' % (h, fails, new_err, len(before)))
    if fails or new_err:
        print('   NOT SAVED: reload %s without saving (or restart the editor) and send i5 this output.' % h); continue
    EA.save_asset(P + h, only_if_is_dirty=False); print('   saved', h, '(backup', dst + ')')
print('total nodes %s: %d' % ('swapped' if APPLY else 'to swap', total))
if not APPLY: print('dry run, nothing changed')
